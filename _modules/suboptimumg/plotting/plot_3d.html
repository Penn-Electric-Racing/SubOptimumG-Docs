def plot3D_surface(
    x_list: NDArray[np.float64],
    y_list: NDArray[np.float64],
    z_list: NDArray[np.float64],
    title: str,
    x_axis: str,
    y_axis: str,
    z_axis: str,
    subtitle: str | None = None,
    theme: str | None = None,
    interp_factor: int = INTERP_FACTOR,
    smoothing_sigma: float = SMOOTHING_SIGMA,
    interp_method: InterpMethod = INTERP_METHOD,
    view_angle_deg: float = VIEW_ANGLE_DEG,
    width: int = PLOT_WIDTH,
    height: int = PLOT_HEIGHT,
) -> go.Figure:
    """
    Plot z as a function of (x, y) as an interactive, rotatable 3D surface with contour lines drawn on it.

    Accepts either grid data or scattered points. Grid data (z is 2D) is interpolated onto a finer grid and drawn as a surface. Scattered data (x, y, z all 1D, same length) is interpolated onto a regular grid to draw the surface, and the original points are overlaid as small black markers.

    Parameters
    ----------
    x_list : NDArray[float64]
        X coordinates. For grid data, the 1D grid values along the x-axis. For scattered data, the x coordinate of each point.
    y_list : NDArray[float64]
        Y coordinates. For grid data, the 1D grid values along the y-axis. For scattered data, the y coordinate of each point.
    z_list : NDArray[float64]
        Z values. For grid data, a 2D array of shape (len(x_list), len(y_list)). For scattered data, a 1D array the same length as x_list and y_list.
    title : str
        Title shown at the top of the figure.
    x_axis : str
        X-axis label, also used in hover text.
    y_axis : str
        Y-axis label, also used in hover text.
    z_axis : str
        Z-axis label, also used as the colorbar title and in hover text.
    subtitle : str | None, optional
        Smaller grey text shown under the title. Default is None (no subtitle).
    theme : str | None, optional
        Name of a color theme from `plotting.color_themes`, which picks the surface colorscale. Default is None (default theme).
    interp_factor : int, optional
        Resample the surface to this many times more points along each axis, making it look smoother. 1 disables interpolation. Default is INTERP_FACTOR (5).
    smoothing_sigma : float, optional
        Gaussian smoothing strength applied to the interpolated surface, in units of interpolated samples. 0 disables smoothing. Default is SMOOTHING_SIGMA (0).
    interp_method : {"linear", "nearest", "slinear", "cubic"}, optional
        Interpolation method for grid data. Scattered data always uses cubic interpolation, falling back to linear. Default is INTERP_METHOD ("linear").
    view_angle_deg : float, optional
        Initial camera angle in degrees, measured around the vertical z-axis from the +x direction. Change this to look at the surface from a different side. Default is VIEW_ANGLE_DEG (-60).
    width : int, optional
        Figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int, optional
        Figure height in pixels. Default is PLOT_HEIGHT (700).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here (fonts, colors, camera distance, aspect ratio) with fig.update_layout, fig.update_traces, or fig.update_scenes before calling fig.show().
    """
    # Detect data type
    data_type = validate_data(x_list, y_list, z_list)

    # Create figure and add trace
    fig = go.Figure()
    theme_colors = get_theme(theme)

    # Prepare smoothed/interpolated data based on input type
    match data_type:
        case DataType3D.GridInput:
            x_dense, y_dense, z_dense = prepare_smooth_data_3D(
                x_list,
                y_list,
                z_list,
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )
        case DataType3D.ScatterInput:
            x_dense, y_dense, z_dense = prepare_smooth_data_3D_scatter(
                x_list,
                y_list,
                z_list,
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )

    # Calculate contour size (use nanmin/nanmax to handle potential NaN values)
    z_min, z_max = np.nanmin(z_dense), np.nanmax(z_dense)
    contour_size = (z_max - z_min) / NUM_CONTOURS if z_max > z_min else 1

    # Add surface plot
    fig.add_trace(
        go.Surface(
            x=x_dense,
            y=y_dense,
            z=z_dense,
            colorscale=theme_colors["colorscale"],
            contours={
                "z": {
                    "show": True,
                    "start": z_min,
                    "end": z_max,
                    "size": contour_size,
                    "width": LINE_WIDTH,
                    "color": CONTOUR_LABEL_COLOR,
                }
            },
            colorbar=dict(
                title=dict(
                    text=z_axis,
                    font=dict(size=FONT_SIZE_MEDIUM),
                ),
                thickness=COLORBAR_THICKNESS,
                len=COLORBAR_LENGTH,
                tickfont=dict(size=FONT_SIZE_SMALL),
            ),
            hovertemplate=(
                f"{x_axis}: %{{x:{FLOAT_PRECISION}}}<br>"
                f"{y_axis}: %{{y:{FLOAT_PRECISION}}}<br>"
                f"{z_axis}: %{{z:{FLOAT_PRECISION}}}<extra></extra>"
            ),
        )
    )

    # Overlay original scatter points if input was scatter data
    if data_type == DataType3D.ScatterInput:
        fig.add_trace(
            go.Scatter3d(
                x=x_list,
                y=y_list,
                z=z_list,
                mode="markers",
                marker=dict(
                    color="black",
                    size=MARKER_SIZE / 2,
                    opacity=0.5,
                ),
                showlegend=False,
                hovertemplate=(
                    f"Original point<br>"
                    f"{x_axis}: %{{x:{FLOAT_PRECISION}}}<br>"
                    f"{y_axis}: %{{y:{FLOAT_PRECISION}}}<br>"
                    f"{z_axis}: %{{z:{FLOAT_PRECISION}}}<extra></extra>"
                ),
            )
        )

    # Configure layout
    full_title = title
    if subtitle is not None:
        full_title += f"<br><span style='font-size: {FONT_SIZE_MEDIUM}px; color: {TEXT_COLOR_LIGHT};'>{subtitle}</span>"

    fig.update_layout(
        title={
            "text": full_title,
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
            "yanchor": TITLE_YANCHOR,
        },
        scene=dict(
            xaxis_title=x_axis,
            yaxis_title=y_axis,
            zaxis_title=z_axis,
            xaxis=dict(
                gridcolor=GRID_COLOR,
                showbackground=True,
                backgroundcolor=SCENE_BGCOLOR,
                tickformat=FLOAT_PRECISION,
            ),
            yaxis=dict(
                gridcolor=GRID_COLOR,
                showbackground=True,
                backgroundcolor=SCENE_BGCOLOR,
                tickformat=FLOAT_PRECISION,
            ),
            zaxis=dict(
                gridcolor=GRID_COLOR,
                showbackground=True,
                backgroundcolor=SCENE_BGCOLOR,
                tickformat=FLOAT_PRECISION,
            ),
            aspectratio=dict(
                x=SCENE_ASPECT_RATIO_X,
                y=SCENE_ASPECT_RATIO_Y,
                z=SCENE_ASPECT_RATIO_Z,
            ),
        ),
        width=width,
        height=height,
        margin=MARGIN,
    )

    fig.update_layout(
        scene_camera=dict(
            center=dict(x=0, y=0, z=0),
            eye=dict(
                x=CAMERA_DISTANCE * np.cos(np.radians(view_angle_deg)),
                y=CAMERA_DISTANCE * np.sin(np.radians(view_angle_deg)),
                z=CAMERA_Z,
            ),
            up=dict(
                x=CAMERA_UP_X,
                y=CAMERA_UP_Y,
                z=CAMERA_UP_Z,
            ),
        )
    )

    return fig

def plot3D_contour(
    x_list: NDArray[np.float64],
    y_list: NDArray[np.float64],
    z_list: NDArray[np.float64],
    title: str,
    x_axis: str,
    y_axis: str,
    z_axis: str,
    subtitle: str | None = None,
    hover_values: NDArray[np.float64] | None = None,
    hover_label: str = "",
    theme: str | None = None,
    interp_factor: int = INTERP_FACTOR,
    smoothing_sigma: float = SMOOTHING_SIGMA,
    interp_method: InterpMethod = INTERP_METHOD,
    log_x: bool = False,
    log_y: bool = False,
    width: int = PLOT_WIDTH,
    height: int = PLOT_HEIGHT,
) -> go.Figure:
    """
    Plot z as a function of (x, y) as a flat, top-down contour map, with z shown by color and labeled contour lines.

    Accepts either grid data or scattered points. Grid data (z is 2D) is interpolated onto a finer grid. Scattered data (x, y, z all 1D, same length) is interpolated onto a regular grid, and the original points are overlaid as black x markers.

    Parameters
    ----------
    x_list : NDArray[float64]
        X coordinates. For grid data, the 1D grid values along the x-axis. For scattered data, the x coordinate of each point.
    y_list : NDArray[float64]
        Y coordinates. For grid data, the 1D grid values along the y-axis. For scattered data, the y coordinate of each point.
    z_list : NDArray[float64]
        Z values shown as color. For grid data, a 2D array of shape (len(x_list), len(y_list)). For scattered data, a 1D array the same length as x_list and y_list.
    title : str
        Title shown at the top of the figure.
    x_axis : str
        X-axis label, also used in hover text.
    y_axis : str
        Y-axis label, also used in hover text.
    z_axis : str
        Name of the z quantity, used as the colorbar title and in hover text.
    subtitle : str | None, optional
        Smaller grey text shown under the title. Default is None (no subtitle).
    hover_values : NDArray[float64] | None, optional
        A second grid of values, same shape as z_list, shown in the hover text alongside z. Grid data only. Default is None (no extra hover values).
    hover_label : str, optional
        Name shown next to hover_values in the hover text. Default is "".
    theme : str | None, optional
        Name of a color theme from `plotting.color_themes`, which picks the colorscale. Default is None (default theme).
    interp_factor : int, optional
        Resample to this many times more points along each axis, making contours smoother. 1 disables interpolation. Default is INTERP_FACTOR (5).
    smoothing_sigma : float, optional
        Gaussian smoothing strength applied to the interpolated grid, in units of interpolated samples. 0 disables smoothing. Default is SMOOTHING_SIGMA (0).
    interp_method : {"linear", "nearest", "slinear", "cubic"}, optional
        Interpolation method for grid data. Scattered data always uses cubic interpolation, falling back to linear. Default is INTERP_METHOD ("linear").
    log_x : bool, optional
        Use a logarithmic x-axis. Default is False (linear).
    log_y : bool, optional
        Use a logarithmic y-axis. Default is False (linear).
    width : int, optional
        Figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int, optional
        Figure height in pixels. Default is PLOT_HEIGHT (700).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here (fonts, colors, contour spacing) with fig.update_layout or fig.update_traces before calling fig.show().
    """
    # Detect data type
    data_type = validate_data(x_list, y_list, z_list)

    theme_colors = get_theme(theme)
    fig = go.Figure()

    # Prepare smoothed/interpolated data based on input type
    match data_type:
        case DataType3D.GridInput:
            x_dense, y_dense, z_dense = prepare_smooth_data_3D(
                x_list,
                y_list,
                z_list,
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )
        case DataType3D.ScatterInput:
            x_dense, y_dense, z_dense = prepare_smooth_data_3D_scatter(
                x_list,
                y_list,
                z_list,
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )
            fig.add_trace(
                go.Scatter(
                    x=x_list,
                    y=y_list,
                    mode="markers",
                    marker=dict(
                        color="black",
                        size=MARKER_SIZE,
                        opacity=0.5,
                        symbol="x",
                    ),
                    showlegend=False,
                    hovertemplate=(
                        f"Original point<br>"
                        f"{x_axis}: %{{x:{FLOAT_PRECISION}}}<br>"
                        f"{y_axis}: %{{y:{FLOAT_PRECISION}}}<br>"
                        f"{z_axis}: %{{customdata:{FLOAT_PRECISION}}}<extra></extra>"
                    ),
                    customdata=z_list,
                )
            )

    # Calculate contour size (use nanmin/nanmax to handle potential NaN values)
    z_min, z_max = np.nanmin(z_dense), np.nanmax(z_dense)
    contour_size = int((z_max - z_min) / NUM_CONTOURS) if z_max > z_min else 1

    hovertemplate = (
        f"{x_axis}: %{{x:{FLOAT_PRECISION}}}<br>"
        f"{y_axis}: %{{y:{FLOAT_PRECISION}}}<br>"
        f"{z_axis}: %{{z:{FLOAT_PRECISION}}}"
    )
    hover_dense = None
    if hover_values is not None:
        _, _, hover_dense = prepare_smooth_data_3D(
            x_list,
            y_list,
            hover_values,
            interp_factor=interp_factor,
            smoothing_sigma=smoothing_sigma,
            interp_method=interp_method,
        )
        hovertemplate += f"<br>{hover_label}: %{{customdata:{FLOAT_PRECISION}}}"

    # Add contour plot
    fig.add_trace(
        go.Contour(
            x=x_dense,
            y=y_dense,
            z=z_dense,
            colorscale=theme_colors["colorscale"],
            contours=dict(
                showlabels=True,
                labelfont=dict(
                    size=FONT_SIZE_SMALL,
                    color=CONTOUR_LABEL_COLOR,
                ),
                start=int(z_min),
                end=int(z_max),
                size=contour_size,
                labelformat=FLOAT_PRECISION,
            ),
            colorbar=dict(
                title=dict(
                    text=z_axis,
                    font=dict(size=FONT_SIZE_MEDIUM),
                ),
                thickness=COLORBAR_THICKNESS,
                len=COLORBAR_LENGTH,
                tickfont=dict(size=FONT_SIZE_SMALL),
                tickformat=FLOAT_PRECISION,
            ),
            customdata=hover_dense,
            hovertemplate=hovertemplate + "<extra></extra>",
        )
    )

    full_title = title
    if subtitle is not None:
        full_title += f"<br><span style='font-size: {FONT_SIZE_MEDIUM}px; color: {TEXT_COLOR_LIGHT};'>{subtitle}</span>"

    fig.update_layout(
        title={
            "text": full_title,
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        xaxis_title={
            "text": x_axis,
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        yaxis_title={
            "text": y_axis,
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        width=width,
        height=height,
        margin=MARGIN,
        plot_bgcolor=PLOT_BGCOLOR,
    )

    fig.update_xaxes(
        tickfont=dict(size=FONT_SIZE_SMALL),
        tickformat=FLOAT_PRECISION,
        type="log" if log_x else "linear",
    )

    fig.update_yaxes(
        tickfont=dict(size=FONT_SIZE_SMALL),
        tickformat=FLOAT_PRECISION,
        type="log" if log_y else "linear",
    )

    return fig
