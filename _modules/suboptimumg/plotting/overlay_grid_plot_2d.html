def overlay_grid_plot_2D(
    x_data_dict: dict[str, dict[str, NDArray[np.number]]],
    y_data_dict: dict[str, dict[str, NDArray[np.number]]],
    subplot_titles: list[str],
    title: str,
    x_label: str,
    y_label: str,
    rows: int,
    cols: int,
    *,
    secondary_y_axis_traces: set[str] = set(),
    secondary_y_axis_label: str = "",
    show_points: bool = False,
    show_legend: bool = False,
    fit_curve: bool = False,
    theme: str | None = None,
    log_x: bool = False,
    log_y: bool = False,
    width: int | None = None,
    height: int | None = None,
    horizontal_spacing: float = SUBPLOT_HORIZONTAL_SPACING,
    vertical_spacing: float = SUBPLOT_VERTICAL_SPACING,
    interp_factor: int = INTERP_FACTOR,
    smoothing_sigma: float = SMOOTHING_SIGMA,
    interp_method: InterpMethod = INTERP_METHOD,
) -> go.Figure:
    """
    Plot a grid of subplots where each subplot overlays the same set of traces, for comparing several data sources
    across categories.

    Parameters
    ----------
    x_data_dict : dict[str, dict[str, NDArray[np.number]]]
        Nested dictionary {subplot_key: {trace_label: x_values}}. Subplots are filled left-to-right, top-to-bottom in key order. Every subplot must contain every trace label.
    y_data_dict : dict[str, dict[str, NDArray[np.number]]]
        Nested dictionary {subplot_key: {trace_label: y_values}} with the same keys as `x_data_dict`. A trace label keeps the same color in every subplot.
    subplot_titles : list[str]
        Title of each subplot, in the same order as the subplot keys.
    title : str
        Overall figure title.
    x_label : str
        X-axis title used on every subplot.
    y_label : str
        Y-axis title used on every subplot.
    rows : int
        Number of subplot rows in the grid.
    cols : int
        Number of subplot columns in the grid.
    secondary_y_axis_traces : set[str], optional
        Trace labels drawn against a second y-axis on the right of each subplot, for series with a different scale. Set its range afterwards with `fig.update_yaxes(range=..., secondary_y=True)`. Default is set() (no secondary axis).
    secondary_y_axis_label : str, optional
        Title of the secondary y-axis. Only used when `secondary_y_axis_traces` is non-empty. Default is "".
    show_points : bool, optional
        If True, draw markers at the original data points. Default is False.
    show_legend : bool, optional
        If True, show a legend with one entry per trace label (clicking an entry toggles that trace in every subplot). Default is False.
    fit_curve : bool, optional
        If True, draw each line as a spline instead of straight segments. Splines need SVG rendering, so this switches traces from WebGL (`go.Scattergl`) to `go.Scatter`, which is slower for long series. Default is False.
    theme : str | None, optional
        Color theme name from `plotting.color_themes`, e.g. "aero". None uses Plotly's qualitative palette. Default is None.
    log_x : bool, optional
        If True, use logarithmic x-axes on all subplots. All x values must be positive. Default is False.
    log_y : bool, optional
        If True, use logarithmic y-axes on all subplots. All y values must be positive. Default is False.
    width : int | None, optional
        Figure width in pixels. None sizes it automatically as max(PLOT_WIDTH, SUBPLOT_WIDTH * cols), i.e. max(1000, 400 * cols). Default is None.
    height : int | None, optional
        Figure height in pixels. None sizes it automatically as max(PLOT_HEIGHT, SUBPLOT_HEIGHT * rows), i.e. max(700, 350 * rows). Default is None.
    horizontal_spacing : float, optional
        Gap between subplot columns, as a fraction of the figure width (0 to 1). Increase if y-axis titles overlap neighbouring subplots. Default is SUBPLOT_HORIZONTAL_SPACING (0.08).
    vertical_spacing : float, optional
        Gap between subplot rows, as a fraction of the figure height (0 to 1). Increase if x-axis titles overlap subplot titles below. Default is SUBPLOT_VERTICAL_SPACING (0.12).
    interp_factor : int, optional
        Each line is resampled to this many times more points. 1 disables interpolation. Default is INTERP_FACTOR (5).
    smoothing_sigma : float, optional
        Gaussian smoothing strength applied to each line, in interpolated samples. 0 disables smoothing. Default is SMOOTHING_SIGMA (0).
    interp_method : {"linear", "nearest", "slinear", "cubic"}, optional
        Interpolation method used when resampling each line. Default is INTERP_METHOD ("linear").

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here (fonts, colors, margins, etc.) with `fig.update_layout(...)` or `fig.update_traces(...)` before calling `fig.show()`.

    Examples
    --------
    >>> fig = overlay_grid_plot_2D(xs, ys, ["Accel", "Skidpad"], "Lap Comparison", "Distance (m)", "Speed (m/s)", rows=1, cols=2, show_legend=True, width=1200)
    """
    # Build figure
    fig = make_subplots(
        rows=rows,
        cols=cols,
        subplot_titles=subplot_titles,
        horizontal_spacing=horizontal_spacing,
        vertical_spacing=vertical_spacing,
        shared_xaxes=False,
        shared_yaxes=False,
        specs=[[{"secondary_y": bool(secondary_y_axis_traces)}] * cols for _ in range(rows)],
    )
    scatter = go.Scatter if fit_curve else go.Scattergl

    subplot_keys = list(y_data_dict.keys())

    # Get all unique trace labels across all subplots
    all_trace_labels: set[str] = set()
    for subplot_key in subplot_keys:
        all_trace_labels.update(y_data_dict[subplot_key].keys())
    trace_labels = sorted(list(all_trace_labels))
    palette = (
        qualitative.Plotly * (len(trace_labels) // len(qualitative.Plotly) + 1)
        if theme is None
        else get_theme_trace_colors(theme, len(trace_labels))
    )
    color_map = {lb: palette[i] for i, lb in enumerate(trace_labels)}

    # Create traces for each subplot
    for subplot_idx, subplot_key in enumerate(subplot_keys):
        r = subplot_idx // cols + 1
        c = subplot_idx % cols + 1

        x_data_for_subplot = x_data_dict[subplot_key]
        y_data_for_subplot = y_data_dict[subplot_key]

        for trace_label in trace_labels:
            color = color_map[trace_label]
            # Get x and y data for this trace
            x_vals = x_data_for_subplot[trace_label]
            y_vals = y_data_for_subplot[trace_label]

            x_dense, y_dense = prepare_smooth_data_2D(
                x_vals.astype(np.float64),
                y_vals.astype(np.float64),
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )

            # Interpolated line
            on_secondary = trace_label in secondary_y_axis_traces
            fig.add_trace(
                scatter(
                    x=x_dense,
                    y=y_dense,
                    mode="lines",
                    line=dict(
                        color=color,
                        width=2,
                        shape=("spline" if fit_curve else "linear"),
                    ),
                    name=trace_label,
                    legendgroup=trace_label,
                    showlegend=(show_legend and subplot_idx == 0),
                ),
                row=r,
                col=c,
                secondary_y=on_secondary,
            )

            # Original points
            if show_points:
                fig.add_trace(
                    scatter(
                        x=x_vals,
                        y=y_vals,
                        mode="markers",
                        marker=dict(color=color, size=MARKER_SIZE, opacity=0.85),
                        name=trace_label,
                        showlegend=False,
                        hovertemplate=f"{x_label}: %{{x}}<br>{subplot_titles[subplot_idx]}: %{{y:.2f}}<extra></extra>",
                    ),
                    row=r,
                    col=c,
                    secondary_y=on_secondary,
                )

        # Axis titles for each subplot
        axis_font = {"size": FONT_SIZE_MEDIUM}
        fig.update_xaxes(title={"text": x_label, "font": axis_font}, row=r, col=c)
        fig.update_yaxes(
            title={"text": y_label, "font": axis_font}, row=r, col=c, secondary_y=False
        )
        if secondary_y_axis_traces:
            fig.update_yaxes(
                title={"text": secondary_y_axis_label, "font": axis_font},
                row=r,
                col=c,
                secondary_y=True,
            )

    # Layout and styling
    fig.update_layout(
        title={
            "text": title,
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        width=width if width is not None else max(PLOT_WIDTH, SUBPLOT_WIDTH * cols),
        height=height if height is not None else max(PLOT_HEIGHT, SUBPLOT_HEIGHT * rows),
        plot_bgcolor=PLOT_BGCOLOR,
        showlegend=show_legend,
        legend=dict(
            orientation="v",
            yanchor="top",
            xanchor="left",
        ),
        margin=MARGIN,
    )

    # Subtle gridlines and zero lines
    fig.update_xaxes(
        type="log" if log_x else "linear",
        showgrid=True,
        gridwidth=1,
        gridcolor="rgba(211, 211, 211, 0.3)",
        zeroline=True,
        zerolinewidth=1,
        zerolinecolor="rgba(211, 211, 211, 0.8)",
    )
    fig.update_yaxes(
        type="log" if log_y else "linear",
        showgrid=True,
        gridwidth=1,
        gridcolor="rgba(211, 211, 211, 0.3)",
        zeroline=True,
        zerolinewidth=1,
        zerolinecolor="rgba(211, 211, 211, 0.8)",
    )

    if secondary_y_axis_traces:
        fig.update_yaxes(showgrid=False, zeroline=False, secondary_y=True)

    return fig
