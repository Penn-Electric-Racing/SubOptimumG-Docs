def plot_track_build_verification(
    distance: NDArray[np.float64],
    curv_raw: NDArray[np.float64],
    curv_filt: NDArray[np.float64],
    radius_filt: NDArray[np.float64],
    x_m: NDArray[np.float64],
    y_m: NDArray[np.float64],
    *,
    width: int = PLOT_WIDTH,
    height: int = PLOT_HEIGHT,
    horizontal_spacing: float = SUBPLOT_HORIZONTAL_SPACING,
) -> go.Figure:
    """
    Side-by-side check of a track built from GPS: the XY track shape (left) and its curvature/radius before and after
    spatial filtering (right).

    Parameters
    ----------
    distance : NDArray[float64]
        Distance along the track for each curvature sample, in meters, on a uniform grid. Same length as `curv_raw`, `curv_filt`, and `radius_filt`.
    curv_raw : NDArray[float64]
        Curvature before filtering, in 1/m. Drawn faded on the left y-axis of the right panel.
    curv_filt : NDArray[float64]
        Curvature after filtering, in 1/m. Drawn on the left y-axis of the right panel.
    radius_filt : NDArray[float64]
        Radius of curvature after filtering, in meters. Drawn on the right y-axis of the right panel.
    x_m : NDArray[float64]
        East (X) coordinates of the track centerline, in meters. Plotted against `y_m` with equal axis scaling.
    y_m : NDArray[float64]
        North (Y) coordinates of the track centerline, in meters. Same length as `x_m`.
    width : int, optional
        Total figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int, optional
        Total figure height in pixels. Default is PLOT_HEIGHT (700).
    horizontal_spacing : float, optional
        Gap between the two panels, as a fraction of the figure width (0 to 1). Default is SUBPLOT_HORIZONTAL_SPACING (0.08).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here with `fig.update_layout(...)` / `fig.update_traces(...)` before calling `fig.show()`.
    """
    fig = make_subplots(
        rows=1,
        cols=2,
        subplot_titles=("Track XY", "Curvature (before/after filter)"),
        horizontal_spacing=horizontal_spacing,
        specs=[[{}, {"secondary_y": True}]],
    )

    # Plot track XY shape
    fig.add_trace(
        go.Scattergl(
            x=x_m,
            y=y_m,
            mode="lines",
            name="Track",
            hovertemplate=f"X: %{{x:{FLOAT_PRECISION}}} m<br>Y: %{{y:{FLOAT_PRECISION}}} m<extra></extra>",
        ),
        row=1,
        col=1,
    )
    fig.update_xaxes(
        title_text="X/East (m)",
        row=1,
        col=1,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
    )
    fig.update_yaxes(
        title_text="Y/North (m)",
        scaleanchor="x",
        scaleratio=1,
        row=1,
        col=1,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
    )

    # Plot raw curvature
    fig.add_trace(
        go.Scattergl(
            x=distance,
            y=curv_raw,
            mode="lines",
            name="Raw Curvature",
            opacity=0.4,
            hovertemplate=f"d: %{{x:{FLOAT_PRECISION}}} m<br>κ: %{{y:{FLOAT_PRECISION}}} 1/m<extra></extra>",
        ),
        row=1,
        col=2,
        secondary_y=False,
    )
    # Plot processed curvature
    fig.add_trace(
        go.Scattergl(
            x=distance,
            y=curv_filt,
            mode="lines",
            name="Filtered Curvature",
            hovertemplate=f"d: %{{x:{FLOAT_PRECISION}}} m<br>κ: %{{y:{FLOAT_PRECISION}}} 1/m<extra></extra>",
        ),
        row=1,
        col=2,
        secondary_y=False,
    )
    fig.update_xaxes(
        title_text="Distance (m)",
        row=1,
        col=2,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
    )
    fig.update_yaxes(
        title_text="Curvature (1/m)",
        row=1,
        col=2,
        secondary_y=False,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
    )

    # Plot processed radius on second y-axis
    fig.add_trace(
        go.Scattergl(
            x=distance,
            y=radius_filt,
            mode="lines",
            name="Filtered Radius",
            hovertemplate=f"d: %{{x:{FLOAT_PRECISION}}} m<br>R: %{{y:{FLOAT_PRECISION}}} m<extra></extra>",
        ),
        row=1,
        col=2,
        secondary_y=True,
    )
    fig.update_yaxes(
        title_text="Radius (m)",
        row=1,
        col=2,
        secondary_y=True,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
    )

    fig.update_layout(
        title={
            "text": "Track Build Verification",
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        height=height,
        width=width,
        margin=MARGIN,
        showlegend=True,
        legend_font=dict(size=FONT_SIZE_SMALL),
        hovermode=HOVER_MODE,
        plot_bgcolor=PLOT_BGCOLOR,
    )
    return fig

def plot_track_heatmap(
    baseline_pos_x: NDArray[np.float64],
    baseline_pos_y: NDArray[np.float64],
    baseline_var: NDArray[np.float64],
    comparison_pos_x: NDArray[np.float64] | None = None,
    comparison_pos_y: NDArray[np.float64] | None = None,
    comparison_var: NDArray[np.float64] | None = None,
    *,
    variable_name: str = "Variable",
    baseline_label: str = "Baseline",
    comparison_label: str = "Comparison",
    data_label: str = "",
    colorscale: str = "Viridis",
    title: str | None = None,
    width: int | None = None,
    height: int = PLOT_HEIGHT,
    horizontal_spacing: float = SUBPLOT_HORIZONTAL_SPACING,
) -> go.Figure:
    """
    2D track heatmap, either single-series or 3-panel comparison.

    When called with only the baseline series (no comparison arguments), a
    single 1:1-aspect scatter plot is returned.  When comparison data is
    supplied, a 3-panel figure is returned: baseline, comparison projected onto the baseline spatial grid, and a signed
    delta panel.

    The comparison series is projected via nearest-neighbour lookup (scipy cKDTree) then linearly interpolated for
    sparse grid points.

    Parameters
    ----------
    baseline_pos_x : NDArray[float64]
        X/East positions of the baseline series, in meters. These positions also define the spatial grid that the comparison is projected onto.
    baseline_pos_y : NDArray[float64]
        Y/North positions of the baseline series, in meters. Same length as `baseline_pos_x`.
    baseline_var : NDArray[float64]
        Value of the variable at each baseline position; sets the marker color. Same length as `baseline_pos_x`.
    comparison_pos_x : NDArray[float64] | None, optional
        X/East positions of the comparison series, in meters. Provide all three comparison arrays to get the 3-panel comparison, or none of them for a single panel. Default is None.
    comparison_pos_y : NDArray[float64] | None, optional
        Y/North positions of the comparison series, in meters. Default is None.
    comparison_var : NDArray[float64] | None, optional
        Value of the variable at each comparison position. Default is None.
    variable_name : str, optional
        Name of the plotted variable. Default is "Variable".
    baseline_label : str, optional
        Title of the baseline panel and name used in hover text. Default is "Baseline".
    comparison_label : str, optional
        Title of the comparison panel and name used in hover text (comparison mode only). Default is "Comparison".
    data_label : str, optional
        Text appended to values in hover text and used as the colorbar title, typically units such as " m/s". Default is "".
    colorscale : str, optional
        Any Plotly named colorscale, e.g. "Viridis", "Plasma", "RdBu". Default is "Viridis".
    title : str | None, optional
        Figure title. None shows no title. Default is None.
    width : int | None, optional
        Total figure width in pixels. None picks PLOT_WIDTH (1000) for a single panel, or SUBPLOT_WIDTH (400) per panel in comparison mode. Default is None.
    height : int, optional
        Total figure height in pixels. Default is PLOT_HEIGHT (700).
    horizontal_spacing : float, optional
        Gap between panels, as a fraction of the figure width (0 to 1). Only matters in comparison mode. Default is SUBPLOT_HORIZONTAL_SPACING (0.08).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here with `fig.update_layout(...)` / `fig.update_traces(...)` before calling `fig.show()`.
    """

    has_comparison = (
        comparison_pos_x is not None and comparison_pos_y is not None and comparison_var is not None
    )

    # Build panel descriptors: (color_data, colorscale, label, hovertemplate, name)
    # The baseline panel is always first; comparison and delta panels are appended when present.
    panels = [
        (
            baseline_var,
            colorscale,
            data_label,
            f"{baseline_label}: %{{customdata:{FLOAT_PRECISION}}}{data_label}<extra></extra>",
            baseline_label,
        )
    ]

    if (
        has_comparison
        and comparison_pos_x is not None
        and comparison_pos_y is not None
        and comparison_var is not None
    ):
        bl_xy = np.column_stack([baseline_pos_x, baseline_pos_y])  # (X, Y)
        tree = cKDTree(bl_xy)
        n_grid = len(bl_xy)

        # Lookup baseline nearest neighbors for comparison points
        cmp_xy = np.column_stack([comparison_pos_x, comparison_pos_y])  # (X, Y)
        _, nearest_idx_raw = tree.query(cmp_xy)
        nearest_idx = np.atleast_1d(nearest_idx_raw)

        # Retrieve comparison variable values on the baseline grid
        vel_sums = np.zeros(n_grid)
        vel_counts = np.zeros(n_grid, dtype=int)
        for i, tidx in enumerate(nearest_idx):
            vel_sums[tidx] += comparison_var[i]
            vel_counts[tidx] += 1

        # Average values for grid points with multiple neighbors, leave empty points as NaN
        has_data = vel_counts > 0
        cmp_projected = np.full(n_grid, np.nan)
        cmp_projected[has_data] = vel_sums[has_data] / vel_counts[has_data]

        # Interpolate linearly over any grid points that have no comparison data
        if int(np.sum(~has_data)) > 0:
            # indices of grid points with data
            valid = np.where(~np.isnan(cmp_projected))[0]
            if 0 < len(valid) < n_grid:
                cmp_projected = np.interp(np.arange(n_grid), valid, cmp_projected[valid])

        delta = baseline_var - cmp_projected
        d_abs = float(max(abs(np.nanmin(delta)), abs(np.nanmax(delta))))

        panels.append(
            (
                cmp_projected,
                colorscale,
                data_label,
                f"{comparison_label}: %{{customdata:{FLOAT_PRECISION}}}{data_label}<extra></extra>",
                comparison_label,
            )
        )
        panels.append(
            (
                delta,
                colorscale,
                f"Delta {data_label}",
                f"Delta: %{{customdata:+{FLOAT_PRECISION}}}{data_label}<extra></extra>",
                "Delta",
            )
        )

    n_cols = len(panels)
    subplot_titles = [baseline_label] + (
        [comparison_label, f"Delta ({baseline_label} - {comparison_label})"]
        if has_comparison
        else []
    )
    fig = make_subplots(
        rows=1,
        cols=n_cols,
        subplot_titles=subplot_titles,
        horizontal_spacing=horizontal_spacing,
    )

    # Helps position colorbars
    col_width = (1.0 - horizontal_spacing * (n_cols - 1)) / n_cols

    for col_idx, (color_data, cs, label, hover, name) in enumerate(panels, start=1):
        subplot_right = col_idx * col_width + (col_idx - 1) * horizontal_spacing
        colorbar_x = subplot_right + 0.01

        fig.add_trace(
            go.Scattergl(
                x=baseline_pos_x,
                y=baseline_pos_y,
                mode="markers",
                marker=dict(
                    size=MARKER_SIZE,
                    color=color_data,
                    colorscale=cs,
                    colorbar=dict(
                        title=dict(
                            text=label,
                            font=dict(size=FONT_SIZE_MEDIUM),
                        ),
                        thickness=COLORBAR_THICKNESS,
                        len=COLORBAR_LENGTH,
                        tickfont=dict(size=FONT_SIZE_SMALL),
                        x=colorbar_x,
                        xanchor="left",
                    ),
                    showscale=True,
                ),
                name=name,
                customdata=color_data,
                showlegend=False,
                hovertemplate=hover,
            ),
            row=1,
            col=col_idx,
        )

    # Batch update axes
    for col_idx in range(1, n_cols + 1):
        ref = f"x{col_idx}" if col_idx > 1 else "x"
        fig.update_xaxes(
            title_text="X/East (m)",
            title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
            row=1,
            col=col_idx,
        )
        fig.update_yaxes(
            title_text="Y/North (m)",
            title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
            scaleanchor=ref,
            scaleratio=1,
            row=1,
            col=col_idx,
        )

    fig.update_layout(
        title={
            "text": title,
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        height=height,
        width=(
            width
            if width is not None
            else (n_cols * SUBPLOT_WIDTH if has_comparison else PLOT_WIDTH)
        ),
        margin=MARGIN,
        showlegend=False,
        hovermode=HOVER_MODE,
        plot_bgcolor=PLOT_BGCOLOR,
    )
    return fig

def plot_lap_overlay(
    laps: list[tuple[NDArray[np.float64], NDArray[np.float64]]],
    baseline_idx: int,
    *,
    sf_line: LineSegment | None = None,
    width: int = PLOT_WIDTH,
    height: int = PLOT_HEIGHT,
) -> go.Figure:
    """
    Overlay the GPS trajectories of multiple laps on one XY plot with equal axis scaling.

    Parameters
    ----------
    laps : list[tuple[NDArray[float64], NDArray[float64]]]
        One `(pos_x, pos_y)` tuple per lap, giving local X/East and Y/North coordinates in meters. Laps are labeled "Lap 1", "Lap 2", ... in list order.
    baseline_idx : int
        Index into `laps` of the lap to highlight with larger markers.
    sf_line : LineSegment | None, optional
        Start/finish line drawn on top, as `((x1, y1), (x2, y2))` in the same local meters as the laps. None draws no line. Default is None.
    width : int, optional
        Figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int, optional
        Figure height in pixels. Default is PLOT_HEIGHT (700).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here with `fig.update_layout(...)` / `fig.update_traces(...)` before calling `fig.show()`.
    """
    fig = go.Figure()

    for i, (pos_x, pos_y) in enumerate(laps):
        fig.add_trace(
            go.Scattergl(
                x=pos_x,
                y=pos_y,
                mode="markers",
                marker=dict(size=MARKER_SIZE_LARGE if i == baseline_idx else MARKER_SIZE),
                name=f"Lap {i + 1}",
                hovertemplate=(
                    f"X: %{{x:{FLOAT_PRECISION}}} m<br>"
                    f"Y: %{{y:{FLOAT_PRECISION}}} m<extra></extra>"
                ),
            )
        )

    if sf_line is not None:
        fig.add_trace(
            go.Scattergl(
                x=[sf_line[0][0], sf_line[1][0]],  # X (East/lon), index 0
                y=[sf_line[0][1], sf_line[1][1]],  # Y (North/lat), index 1
                mode="lines+markers",
                line=dict(width=LINE_WIDTH),
                marker=dict(size=MARKER_SIZE_LARGE),
                name="S/F Line",
            )
        )

    fig.update_yaxes(scaleanchor="x", scaleratio=1)
    fig.update_layout(
        title={
            "text": "Lap Overlay",
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        xaxis_title={
            "text": "X (m)",
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        yaxis_title={
            "text": "Y (m)",
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        width=width,
        height=height,
        margin=MARGIN,
        hovermode=HOVER_MODE,
        plot_bgcolor=PLOT_BGCOLOR,
    )

    return fig

def plot_time_delta_comparison_traces(
    laps: dict[
        str,
        tuple[
            NDArray[np.float64],  # distance (m)
            NDArray[np.float64],  # variable values
            NDArray[np.float64],  # velocity (m/s) for time-delta
        ],
    ],
    *,
    variable_name: str = "Variable",
    reference_lap: str | None = None,
    data_label: str = "",
    dx: float = 0.1,
    log_x: bool = False,
    log_y: bool = False,
    width: int = PLOT_WIDTH,
    height: int | None = None,
    vertical_spacing: float = SUBPLOT_VERTICAL_SPACING,
) -> go.Figure:
    """
    2-subplot comparison traces. Left subplot shows the change of a variable over distance along the track, and right
    subplots shows the cumulative time delta relative to a reference lap.

    Parameters
    ----------
    laps : dict[str, tuple[NDArray[float64], NDArray[float64], NDArray[float64]]]
        Maps each lap label to `(distance, variable, velocity)`: three equal-length arrays of cumulative distance (m), the variable to plot, and velocity (m/s) used to compute time deltas.
    variable_name : str, optional
        Name of the plotted variable, used in the title and top y-axis label. Default is "Variable".
    reference_lap : str | None, optional
        Key from `laps` to measure time deltas against. None picks the lap with the lowest estimated total time. Default is None.
    data_label : str, optional
        Text appended to values in hover text, typically units such as " m/s". Default is "".
    dx : float, optional
        Step of the common distance grid that all laps are resampled onto, in meters. Default is 0.1.
    log_x : bool, optional
        Use a logarithmic distance (x) axis. Default is False.
    log_y : bool, optional
        Use logarithmic y-axes on both subplots. Default is False.
    width : int, optional
        Figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int | None, optional
        Total figure height in pixels. None uses SUBPLOT_HEIGHT (350) per subplot, i.e. 700. Default is None.
    vertical_spacing : float, optional
        Gap between the two subplots, as a fraction of the figure height (0 to 1). Default is SUBPLOT_VERTICAL_SPACING (0.12).

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here with `fig.update_layout(...)` / `fig.update_traces(...)` before calling `fig.show()`.
    """
    # Build common distance grid
    all_max_dists = [v[0][-1] for v in laps.values()]
    grid_max = min(all_max_dists)
    grid = np.arange(0, grid_max, dx)

    # Interpolate every series onto the grid
    grid_var: dict[str, NDArray[np.float64]] = {}
    grid_vel: dict[str, NDArray[np.float64]] = {}
    for k, v in laps.items():
        d, var, vel = v
        grid_var[k] = np.interp(grid, d, var, left=np.nan, right=np.nan)
        grid_vel[k] = np.interp(grid, d, vel, left=np.nan, right=np.nan)

    # Use specified reference lap or select the one with the lowest estimated total drive time
    if reference_lap is not None:
        if reference_lap not in laps:
            raise ValueError(f"reference_lap={reference_lap} not in laps")
        ref_label = reference_lap
    else:
        ref_label = None
        ref_time = float("inf")
        for k in laps.keys():
            # Estimate by summing dt = dx / vel, ignoring zero/near-zero velocities
            vel = grid_vel[k]
            dt = np.zeros_like(vel, dtype=float)
            mask = vel > 0.5
            dt[mask] = dx / vel[mask]
            total_drive_time = float(np.nansum(dt))

            if total_drive_time < ref_time:
                ref_time = total_drive_time
                ref_label = k

        if ref_label is None:
            raise ValueError("laps must contain at least one entry")

    # Color map for consistency across subplots and hover templates
    palette = qualitative.Plotly * (len(laps) // len(qualitative.Plotly) + 1)
    color_map: dict[str, str] = {lb: palette[i] for i, lb in enumerate(laps.keys())}

    subplot_titles = (
        f"{variable_name} vs Distance",
        f"Cumulative Time Delta v.s. {ref_label} Lap",
    )
    fig = make_subplots(
        rows=2,
        cols=1,
        subplot_titles=subplot_titles,
        vertical_spacing=vertical_spacing,
        shared_xaxes=True,
    )

    # Variable traces
    for label in laps.keys():
        fig.add_trace(
            go.Scattergl(
                x=grid,
                y=grid_var[label],
                mode="lines",
                name=f"{label} (ref)" if label == ref_label else label,
                legendgroup=label,
                line=dict(
                    color=color_map[label],
                    width=LINE_WIDTH * 2 if label == ref_label else LINE_WIDTH,
                ),
                hovertemplate=f"{label}: %{{y:{FLOAT_PRECISION}}}{data_label}<extra></extra>",
            ),
            row=1,
            col=1,
        )
    fig.update_yaxes(
        title_text=variable_name,
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        row=1,
        col=1,
    )

    if ref_label is None:
        raise ValueError("laps must contain at least one entry")

    # Cumulative time delta vs reference
    ref_vel = grid_vel[ref_label]
    dt_ref = np.zeros_like(ref_vel, dtype=float)
    mask = ref_vel > 0.5
    dt_ref[mask] = dx / ref_vel[mask]

    for label in laps.keys():
        if label == ref_label:
            continue

        vel = grid_vel[label]
        dt_cmp = np.zeros_like(vel, dtype=float)
        mask = vel > 0.5
        dt_cmp[mask] = dx / vel[mask]
        cum_delta = np.cumsum(dt_cmp - dt_ref)

        fig.add_trace(
            go.Scattergl(
                x=grid,
                y=cum_delta,
                mode="lines",
                name=f"Delta t of {label}",
                legendgroup=label,
                showlegend=False,
                line=dict(color=color_map[label], width=LINE_WIDTH),
                hovertemplate=(
                    f"{label} %{{y:{FLOAT_PRECISION}}}s vs {ref_label}<br>"
                    f"x=%{{x:{FLOAT_PRECISION}}}<extra></extra>"
                ),
            ),
            row=2,
            col=1,
        )
    fig.update_xaxes(
        title_text="Distance (m)",
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        row=2,
        col=1,
    )
    fig.update_yaxes(
        title_text="Delta t (s) (positive = slower)",
        title_font=dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        row=2,
        col=1,
    )

    fig.update_layout(
        title={
            "text": f"{variable_name} Comparison (ref: {ref_label})",
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
