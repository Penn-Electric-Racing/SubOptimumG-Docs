def overlay_plot2D(
    x_data_dict: dict[str, NDArray[np.number]],
    y_data_dict: dict[str, NDArray[np.number]],
    title: str,
    x_axis: str,
    y_axis: str,
    *,
    h_line: float | None = None,
    h_line_label: str | None = None,
    v_line: float | None = None,
    v_line_label: str | None = None,
    fit_curve: bool = False,
    show_points: bool = False,
    subtitle: str | None = None,
    theme: str | None = None,
    log_x: bool = False,
    log_y: bool = False,
    width: int = PLOT_WIDTH,
    height: int = PLOT_HEIGHT,
    interp_factor: int = INTERP_FACTOR,
    smoothing_sigma: float = SMOOTHING_SIGMA,
    interp_method: InterpMethod = INTERP_METHOD,
) -> go.Figure:
    """
    Plot several y-vs-x series on the same axes, one trace per dictionary key.

    Parameters
    ----------
    x_data_dict : dict[str, NDArray[np.number]]
        Maps each trace label (shown in the legend) to that trace's x values. Must have the same keys as `y_data_dict`.
    y_data_dict : dict[str, NDArray[np.number]]
        Maps each trace label to that trace's y values. Traces are drawn in sorted key order.
    title : str
        Plot title shown at the top of the figure.
    x_axis : str
        X-axis title, e.g. "Mass (kg)". Also used in hover text.
    y_axis : str
        Y-axis title, e.g. "Points". Also used in hover text.
    h_line : float | None, optional
        Y value at which to draw a horizontal dashed reference line across all traces. None draws no line. Default is None.
    h_line_label : str | None, optional
        Legend label for the horizontal reference line. Default is None.
    v_line : float | None, optional
        X value at which to draw a vertical dashed reference line across all traces. None draws no line. Default is None.
    v_line_label : str | None, optional
        Legend label for the vertical reference line. Default is None.
    fit_curve : bool, optional
        If True, draw interpolated (and optionally smoothed) curves through each trace's points. Controlled by `interp_factor`, `smoothing_sigma`, and `interp_method`. Default is False.
    show_points : bool, optional
        If True, draw markers at the original data points. Default is False.
    subtitle : str | None, optional
        Smaller grey text shown under the title. Default is None.
    theme : str | None, optional
        Color theme name from `plotting.color_themes`, e.g. "aero". None uses Plotly's default trace colors. Default is None.
    log_x : bool, optional
        If True, use a logarithmic x-axis. All x values must be positive. Default is False.
    log_y : bool, optional
        If True, use a logarithmic y-axis. All y values must be positive. Default is False.
    width : int, optional
        Figure width in pixels. Default is PLOT_WIDTH (1000).
    height : int, optional
        Figure height in pixels. Default is PLOT_HEIGHT (700).
    interp_factor : int, optional
        Only used when `fit_curve` is True. Resample each trace to this many times more points. 1 disables interpolation. Default is INTERP_FACTOR (5).
    smoothing_sigma : float, optional
        Only used when `fit_curve` is True. Gaussian smoothing strength in interpolated samples. 0 disables smoothing. Default is SMOOTHING_SIGMA (0).
    interp_method : {"linear", "nearest", "slinear", "cubic"}, optional
        Only used when `fit_curve` is True. Interpolation method. Default is INTERP_METHOD ("linear").

    Returns
    -------
    go.Figure
        Plotly figure. Customize anything not exposed here (fonts, colors, margins, etc.) with `fig.update_layout(...)` or `fig.update_traces(...)` before calling `fig.show()`.

    Examples
    --------
    >>> fig = overlay_plot2D({"A": x_a, "B": x_b}, {"A": y_a, "B": y_b}, "Comparison", "Mass (kg)", "Points", log_y=True)
    """

    fig = go.Figure()

    trace_labels = sorted(list(y_data_dict.keys()))
    trace_colors: list[str | None] = (
        [None] * len(trace_labels)
        if theme is None
        else list(get_theme_trace_colors(theme, len(trace_labels)))
    )

    # Create traces for each series
    for trace_label, trace_color in zip(trace_labels, trace_colors):
        x_vals = x_data_dict[trace_label]
        y_vals = y_data_dict[trace_label]

        if fit_curve:
            x_dense, y_dense = prepare_smooth_data_2D(
                x_vals.astype(np.float64),
                y_vals.astype(np.float64),
                interp_factor=interp_factor,
                smoothing_sigma=smoothing_sigma,
                interp_method=interp_method,
            )
        else:
            x_dense, y_dense = x_vals.astype(np.float64), y_vals.astype(np.float64)

        # Interpolated line
        fig.add_trace(
            go.Scatter(
                x=x_dense,
                y=y_dense,
                mode="lines+markers" if show_points else "lines",
                line=dict(width=LINE_WIDTH, shape="spline", color=trace_color),
                marker=dict(size=MARKER_SIZE_LARGE) if show_points else None,
                name=trace_label,
                hovertemplate=f"{x_axis}: %{{x:{FLOAT_PRECISION}}}<br>{y_axis}: %{{y:{FLOAT_PRECISION}}}<extra></extra>",
            )
        )

    reference_line_style = dict(
        color=REFERENCE_LINE_COLOR,
        width=REFERENCE_LINE_WIDTH,
        dash=REFERENCE_LINE_DASH,
    )
    all_x = np.concatenate([x_data_dict[label] for label in trace_labels])
    all_y = np.concatenate([y_data_dict[label] for label in trace_labels])

    if h_line is not None:
        fig.add_trace(
            go.Scatter(
                x=[min(all_x), max(all_x)],
                y=[h_line, h_line],
                mode="lines",
                line=reference_line_style,
                name=f"{h_line_label}",
                hoverinfo="name",
            )
        )

    if v_line is not None:
        fig.add_trace(
            go.Scatter(
                x=[v_line, v_line],
                y=[min(all_y), max(all_y)],
                mode="lines",
                line=reference_line_style,
                name=f"{v_line_label}",
                hoverinfo="name",
            )
        )

    # Title, Axes, Label, Legend, Size
    full_title = title
    if subtitle is not None:
        full_title += f"<br><span style='font-size: {FONT_SIZE_MEDIUM}px; color: {TEXT_COLOR_LIGHT};'>{subtitle}</span>"

    fig.update_layout(
        title={
            "text": full_title,
            "font": dict(size=FONT_SIZE_LARGE, color=TEXT_COLOR_DARK),
            "x": TITLE_X,
            "xanchor": TITLE_XANCHOR,
        },
        xaxis_title={
            "text": x_axis,
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        yaxis_title={
            "text": y_axis,
            "font": dict(size=FONT_SIZE_MEDIUM, color=TEXT_COLOR_DARK),
        },
        legend_title="Legend",
        showlegend=True,
        legend_title_font=dict(size=FONT_SIZE_MEDIUM),
        legend_font=dict(size=FONT_SIZE_SMALL),
        hovermode=HOVER_MODE,
        plot_bgcolor=PLOT_BGCOLOR,
        width=width,
        height=height,
        margin=MARGIN,
    )

    # Add subtle grid lines
    fig.update_xaxes(
        type="log" if log_x else "linear",
        showgrid=True,
        gridwidth=GRID_WIDTH,
        gridcolor=GRID_COLOR,
        zeroline=True,
        zerolinewidth=ZEROLINE_WIDTH,
        zerolinecolor=ZEROLINE_COLOR,
        tickfont=dict(size=FONT_SIZE_SMALL),
        tickformat=FLOAT_PRECISION,
    )

    fig.update_yaxes(
        type="log" if log_y else "linear",
        showgrid=True,
        gridwidth=GRID_WIDTH,
        gridcolor=GRID_COLOR,
        zeroline=True,
        zerolinewidth=ZEROLINE_WIDTH,
        zerolinecolor=ZEROLINE_COLOR,
        tickfont=dict(size=FONT_SIZE_SMALL),
        tickformat=FLOAT_PRECISION,
    )

    if h_line is not None:
        fig.update_yaxes(
            range=_padded_range(min(min(all_y), h_line), max(max(all_y), h_line), log_y)
        )

    if v_line is not None:
        fig.update_xaxes(
            range=_padded_range(min(min(all_x), v_line), max(max(all_x), v_line), log_x)
        )

    return fig
