class MotorModel(BaseModel):
    """Electric motor configuration."""

    model_config = ConfigDict(validate_assignment=True)

    name: str = Field(description="Product name")
    setup: DriveSetup = Field(description="Drive configuration")
    max_rpm: float = Field(gt=0, description="Max RPM of motor")
    fw_rpm: float = Field(gt=0, description="Field weakening RPM inflection point")
    max_torque: float = Field(gt=0, description="Max torque of motor (Nm)")
    fw_torque: float = Field(gt=0, description="Field weakening torque inflection point (Nm)")
    pow_lim: float = Field(gt=0, description="Power limit (W)")
    moi_motor: float = Field(
        description="MOI (kg-m^2) of motor rotor, with no gearbox accomodations."
    )
    efficiency_method: Literal["sys", "indiv"] = Field(
        description="Efficiency calculation method: 'sys' or 'indiv'"
    )
    moc_efficiency: float = Field(gt=0, le=1.0, description="Motor controller efficiency")
    motor_efficiency: float = Field(gt=0, le=1.0, description="Motor efficiency")
    chain_efficiency: float = Field(gt=0, le=1.0, description="Chain efficiency")
    diff_efficiency: float = Field(
        gt=0, le=1.0, description="Differential efficiency (currently unused)"
    )
    system_efficiency: float = Field(
        gt=0,
        le=1.0,
        description="Alternative to product of individual efficiencies if efficiency_method is 'sys'",
    )
    mass: float = Field(gt=0, description="Motor mass (kg)")
    ideal_gear_ratio: float = Field(gt=0, description="Ideal gear ratio (currently unused)")
    use_efficiency_lut: bool = Field(
        description="Whether or not to initialize a more advanced motor efficiency solution"
    )
    efficiency_data: list[list[float]] = Field(
        description="List of efficiency map data points [RPM, Torque (Nm), Efficiency (%)]"
    )

    @field_validator("efficiency_data")
    def validate_efficiency_data_plausible(cls, v: list[list[float]]) -> list[list[float]]:
        if len(v) < 25:
            print(
                f"WARNING: Efficiency data {v} must contain at least 25 unique points for correct fitment."
            )
        else:
            for i, data in enumerate(v):
                if len(data) != 3:
                    print(
                        f"WARNING: Efficiency data points {data} must have 3 elements [RPM, Torque (Nm), Efficiency (%)]. Error at idx: {i}."
                    )
                if data[2] < 0.7 or data[2] > 1.00:
                    print(
                        f"WARNING: Efficiency data point idx:{i} has efficiency {data[2]} outside recommended range [0.70, 1.00]"
                    )
        return v

    @model_validator(mode="after")
    def validate_regen_config(self) -> PowertrainModel:
        if self.regen_percent == 0 and self.max_regen > 0:
            print(
                "WARNING: Max regen is non-zero but regen percent is zero. This will result in no regen braking"
            )
        if self.regen_percent == 1 and self.max_regen <= 0:
            print(
                "WARNING: Regen percent is 1 but max regen is not specified. This will result in no regen braking"
            )
        return self

    @field_validator("tire_radius")
    def validate_tire_radius_plausible(cls, v: float) -> float:
        """Check tire diameter (converted from inches) against a plausible range.

        Notes
        -----
        ``tire_radius`` is in inches; the diameter is converted to meters
        before comparing against the recommended 8-24 inch range.
        """
        diameter_m = in_to_m(v * 2)
        min_diam_m = in_to_m(8)
        max_diam_m = in_to_m(24)
        if not (min_diam_m <= diameter_m <= max_diam_m):
            print(
                f"WARNING: Tire diameter of {diameter_m:.4f} m is outside recommended range [{min_diam_m:.4f}, {max_diam_m:.4f}] m (8-24 inches)"
            )
        return v
