class SweepParamConfig(BaseModel):

    model_config = ConfigDict(frozen=True, extra="forbid")

    name: str = Field(description="Name of the parameter")
    min: float | int = Field(description="Lower bound")

class SweepData1D(BaseModel):
    """Strongly typed 1D sweep data container."""

    var_name: str = Field(description="Variable name being swept")
    sweep_values: NDArray[np.float64] = Field(
        description="Actual parameter values used in the sweep (x-values)"
    )

    accel_pts: NDArray[np.float64] = Field(description="Acceleration event points")
    skidpad_pts: NDArray[np.float64] = Field(description="Skidpad event points")
    autoX_pts: NDArray[np.float64] = Field(description="Autocross event points")
    endurance_pts: NDArray[np.float64] = Field(description="Endurance event points")
    efficiency_pts: NDArray[np.float64] = Field(description="Efficiency points")
    accel_t: NDArray[np.float64] = Field(description="Acceleration event time")
    skidpad_t: NDArray[np.float64] = Field(description="Skidpad event time")
    autoX_t: NDArray[np.float64] = Field(description="Autocross event time")
    endurance_t: NDArray[np.float64] = Field(description="Endurance event time")

    plausible: NDArray[np.bool_] = Field(
        description="Plausibility check result for each sweep step"
    )

    model_config = ConfigDict(arbitrary_types_allowed=True, frozen=True)

    @computed_field  # type: ignore[prop-decorator]
    # See vehicle_models.py: valid Pydantic V2 pattern, mypy plugin currently
    # incompatible with the installed mypy version.
    @property
    def total_pts(self) -> NDArray[np.float64]:
        """Compute total points as sum of all point categories."""
        return (
            self.accel_pts
            + self.skidpad_pts
            + self.autoX_pts
            + self.endurance_pts
            + self.efficiency_pts
        )

    @classmethod
    def create(cls, var_name: str, sweep_values: NDArray[np.float64]) -> SweepData1D:
        """Create a SweepData1D with pre-allocated numpy arrays of specified length."""
        length = len(sweep_values)
        return cls(

            sweep_values=sweep_values,
            accel_pts=np.zeros(length, dtype=float),
            skidpad_pts=np.zeros(length, dtype=float),
            autoX_pts=np.zeros(length, dtype=float),
            endurance_pts=np.zeros(length, dtype=float),

            skidpad_t=np.zeros(length, dtype=float),
            autoX_t=np.zeros(length, dtype=float),
            endurance_t=np.zeros(length, dtype=float),
            plausible=np.zeros(length, dtype=bool),
        )

    def check_values_are_settable(self, vehicle: VehicleModel) -> None:
        """Ensure the name is a car parameter that accepts every value of the axis."""
        for value in self.values:
            try:
                vehicle.set_parameter(self.name, float(value))
            except (AttributeError, ValidationError) as error:
                raise ValueError(f"Can't sweep `{self.name}` to {value:.4g}: {error}") from error

    @model_validator(mode="after")
    def check_unique_names(self) -> SweepGrid:
        """Ensure no parameter is swept more than once."""
        names = [axis.name for axis in self.axes]
        for name in names:
            if names.count(name) > 1:
                raise ValueError(f"`{name}` is swept more than once")
        return self





[docs]
    def check_values_are_settable(self, base_vehicle: VehicleModel) -> None:
        """
        Ensure every axis accepts every one of its values, before any simulation runs.

        Parameters
        ----------
        base_vehicle : VehicleModel
            The vehicle the values will be applied to. It is not modified.

        Raises
        ------
        ValueError
            If a name is not a parameter, or a value fails the parameter's validation
        """
        vehicle = base_vehicle.model_copy(deep=True)
        for axis in self.axes:
            axis.check_values_are_settable(vehicle)

    def iterate_grid_points(self) -> Iterator[list[ParameterModification]]:
        """Yield the swept values of each grid point"""
        for point in itertools.product(*(axis.values for axis in self.axes)):
            yield [
                ParameterModification(name=axis.name, value=float(value))
                for axis, value in zip(self.axes, point)
            ]

    )



[docs]
    @model_validator(mode="after")
    def check_configuration_count(self) -> SweepPlan:
        """Ensure there is exactly one configuration per grid point."""
        expected = math.prod(len(axis.values) for axis in self.axes)
        if len(self.configurations) != expected:
            raise ValueError(
                f"Expected {expected} configurations for the grid, got {len(self.configurations)}"
            )
        return self
