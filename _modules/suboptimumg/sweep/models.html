class CategoricalSweepInput(BaseModel):
    """
    Fields needed to sweep one car entry across labeled replacement values.

    Notes
    -----
    Each option replaces exactly one entry, the one at `name`. The entry can be a whole sub-model
    (e.g. 'pwrtn.motor') or a single value (e.g. a flag).
    """

    model_config = ConfigDict(frozen=True, extra="forbid")

    name: str = Field(
        description="Dotted name of the entry each option replaces, e.g. 'pwrtn.motor'"
    )
    options: dict[str, CategoricalValue] = Field(
        min_length=1, description="Replacement value for the entry, keyed by label"
    )

class SweepData1D(BaseModel):
    """Strongly typed 1D sweep data container."""

    var_name: str = Field(description="Variable name being swept")
    sweep_values: NDArray[np.float64] = Field(
        description="Actual parameter values used in the sweep (x-values)"
    )

    accel_pts: NDArray[np.float64] = Field(description="Acceleration event points")
    skidpad_pts: NDArray[np.float64] = Field(description="Skidpad event points")
    autoX_pts: NDArray[np.float64] = Field(description="Autocross event points")
    endurance_pts: NDArray[np.float64] = Field(description="Endurance event points")
    efficiency_pts: NDArray[np.float64] = Field(description="Efficiency points")
    accel_t: NDArray[np.float64] = Field(description="Acceleration event time")
    skidpad_t: NDArray[np.float64] = Field(description="Skidpad event time")
    autoX_t: NDArray[np.float64] = Field(description="Autocross event time")
    endurance_t: NDArray[np.float64] = Field(description="Endurance event time")

    plausible: NDArray[np.bool_] = Field(
        description="Plausibility check result for each sweep step"
    )

    model_config = ConfigDict(arbitrary_types_allowed=True, frozen=True)

    @computed_field  # type: ignore[prop-decorator]
    # See vehicle_models.py: valid Pydantic V2 pattern, mypy plugin currently
    # incompatible with the installed mypy version.
    @property
    def total_pts(self) -> NDArray[np.float64]:
        """Compute total points as sum of all point categories."""
        return (
            self.accel_pts
            + self.skidpad_pts
            + self.autoX_pts
            + self.endurance_pts
            + self.efficiency_pts
        )

    def try_get_index_of_label(self, label: float | str) -> int:
        if not isinstance(label, str) or label not in self.labels:

class SweepGrid(BaseModel):
    """The swept axes, whose every combination of values is one grid point."""

    model_config = ConfigDict(frozen=True)

    axes: list[SweepAxis] = Field(min_length=1, description="Swept axes, in input order")
    modifications_for_each_axis_value: list[list[Modification]] = Field(
        description="For each axis, the modification that sets each of its values, in the axis's order"
    )



[docs]
    @classmethod
    def from_inputs(cls, inputs: Sequence[SweepInput]) -> SweepGrid:
        axes: list[NumericAxis | CategoricalAxis] = []
        modifications_for_each_axis_value: list[list[Modification]] = []
        for sweep_input in inputs:
            match sweep_input:
                case SweepParamConfig():
                    numeric_axis = NumericAxis.from_sweep_param(sweep_input)
                    axes.append(numeric_axis)
                    modifications_for_each_axis_value.append(
                        numeric_axis.modifications_for_each_value()
                    )
                case CategoricalSweepInput():
                    axes.append(
                        CategoricalAxis(name=sweep_input.name, labels=list(sweep_input.options))
                    )
                    modifications_for_each_axis_value.append(
                        [
                            CategoricalModification(name=sweep_input.name, label=label, value=value)
                            for label, value in sweep_input.options.items()
                        ]
                    )
        return cls(axes=axes, modifications_for_each_axis_value=modifications_for_each_axis_value)

    @model_validator(mode="after")
    def check_unique_names(self) -> SweepGrid:
        """Ensure no parameter is swept more than once."""
        names = [axis.name for axis in self.axes]
        for name in names:
            if names.count(name) > 1:
                raise ValueError(f"`{name}` is swept more than once")
        return self

    def check_values_are_settable(self, base_vehicle: VehicleModel) -> None:
        """
        Ensure every axis accepts every one of its values, before any simulation runs.

        Parameters
        ----------
        base_vehicle : VehicleModel
            The vehicle the values will be applied to. It is not modified.

        Raises
        ------
        ValueError
            If a name is not a parameter, or a value fails the parameter's validation
        """
        for axis_value_modifications in self.modifications_for_each_axis_value:
            for modification in axis_value_modifications:
                vehicle = base_vehicle.model_copy(deep=True)
                try:
                    modification.apply_to_vehicle(vehicle)
                except (AttributeError, ValidationError) as error:
                    raise ValueError(f"Can't sweep `{modification.name}`: {error}") from error

    def iterate_grid_points(self) -> Iterator[list[Modification]]:
        """
        Yield the swept modifications of each grid point. For each list of modifications,
        categorical comes before numerical.
        """
        for point in itertools.product(*self.modifications_for_each_axis_value):
            yield sorted(point, key=lambda modification: modification.kind != "categorical")

[docs]
    @model_validator(mode="after")
    def check_configuration_count(self) -> SweepPlan:
        """Ensure there is exactly one configuration per grid point."""
        expected = math.prod(axis.size for axis in self.axes)
        if len(self.configurations) != expected:
            raise ValueError(
                f"Expected {expected} configurations for the grid, got {len(self.configurations)}"
            )
        return self
