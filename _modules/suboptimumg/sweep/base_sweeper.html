class BaseSweeper(ABC, Generic[PlanT, DataT]):
    """
    Owns a base competition and the results of sweeping it.

    Parameters
    ----------
    comp_data : CompetitionData
        Base competition. It is copied, so later edits to the original have no effect.
    executor : SweepExecutor
        How simulations are run

    Notes
    -----
    Subclasses say what to simulate in `_generate_plan`, and how to arrange the results in `_assemble_data`. The plan is
    regenerated on every `sweep()`, so edits to `comp_data` or the subclass's configuration are picked up. The
    executor's simulator is part of the fingerprint, so changing what one simulation means also reruns.
    """

    def __init__(self, comp_data: CompetitionData, executor: SweepExecutor) -> None:
        self.comp_data = comp_data.model_copy(deep=True)
        self.executor = executor
        self._data: DataT | None = None
        self._fingerprint: str | None = None

    def sweep(self, force: bool = False) -> None:
        """
        Run every simulation of the sweep, unless the results are already up to date.

        Parameters
        ----------
        force : bool, optional
            Rerun even when the results are up to date (default is False)
        """
        plan = self._generate_plan()
        fingerprint = self._compute_plan_fingerprint(plan)
        if not force and fingerprint == self._fingerprint:
            print("Sweep results are up to date; skipping the simulations.")
            return

        results = self.executor.simulate_configurations(self.comp_data, plan.configurations)
        self._data = self._assemble_data(plan, results)
        self._fingerprint = fingerprint



    @property
    def is_stale(self) -> bool:
        """Whether the configuration has changed since the last sweep, or none has run."""
        return self._fingerprint != self._compute_plan_fingerprint(self._generate_plan())

    @property
    def data(self) -> DataT:
        """Results of the last sweep."""
        if self._data is None:
            raise RuntimeError("No results yet. Call sweep() first.")
        if self.is_stale:
            print(
                "The sweep configuration has changed since the last sweep(). "
                "These results are out of date; call sweep() to refresh them."
            )
        return self._data

    @abstractmethod
    def _generate_plan(self) -> PlanT: ...

    @abstractmethod
    def _assemble_data(self, plan: PlanT, results: list[ConfigurationResult]) -> DataT: ...

    def _compute_plan_fingerprint(self, plan: PlanT) -> str:
        return hashlib.sha256(
            pickle.dumps((self.comp_data, plan.configurations, self.executor.simulator))
        ).hexdigest()
