class ArraySweeper(BaseSweeper[ArraySweepPlan, ArraySweepData]):
    """
    Sweeps each parameter on its own, across percent changes from its base value.

    Parameters
    ----------
    comp_data : CompetitionData
        Base competition. It is copied, so later edits to the original have no effect.
    parameter_names : Sequence[str]
        Numeric parameters to sweep, e.g. `["mass", "aero.cl"]`
    percent_changes : Sequence[float], optional
        Changes from each parameter's base value, as ascending fractions
        (default is -10% to +10% in steps of 2%)
    executor : SweepExecutor, optional
        How simulations are run (default runs them in a process pool)

    Notes
    -----
    Every parameter's simulations, plus one of the unmodified car for the baseline, run in a single batch.
    `comp_data`, `parameter_names` and `percent_changes` can be edited after construction. `sweep()` only
    reruns the simulations when the generated configurations have changed.
    """

    def __init__(
        self,
        comp_data: CompetitionData,
        parameter_names: Sequence[str],
        percent_changes: Sequence[float] = DEFAULT_PERCENT_CHANGES,
        executor: SweepExecutor = MultiprocessSweepExecutor(),
    ) -> None:
        super().__init__(comp_data, executor)
        self.parameter_names = list(parameter_names)
        self.percent_changes = list(percent_changes)

    def plot(
        self,
        metric: SweepMetric = SweepMetric.TOTAL_PTS,
        *,
        title: str | None = None,
        y_label: str | None = None,
        **plot_options: object,
    ) -> go.Figure:
        """
        Plot one metric of every swept parameter, colored by percent change.

        Parameters
        ----------
        metric : SweepMetric, optional
            Metric to plot (default is SweepMetric.TOTAL_PTS)
        title, y_label : str, optional
            Text for the plot. Defaults are generated from the metric.
        **plot_options : object
            Passed to the plotting function, e.g. `theme="powertrain"`

        Returns
        -------
        go.Figure
            One column of points per parameter, with the unmodified car's value as a baseline
        """
        return plot_array_sweep(
            self.data, metric=metric, title=title, y_label=y_label, **plot_options
        )



    def _generate_plan(self) -> ArraySweepPlan:
        base_vehicle = self.comp_data.vehicle_model
        return ArraySweepPlan(
            percent_changes=self.percent_changes,
            baseline=SweepPlan(
                axes=[], configurations=[CarConfiguration(modifications=[])], dependency_names=[]
            ),
            parameter_plans={
                name: SweepConfigGenerator(
                    inputs=[RelativeSweepInput(name=name, percent_changes=self.percent_changes)]
                ).generate_sweep_plan(base_vehicle)
                for name in self.parameter_names
            },
        )

    def _assemble_data(
        self, plan: ArraySweepPlan, results: list[ConfigurationResult]
    ) -> ArraySweepData:
        return ArraySweepData.from_results(plan, results)
