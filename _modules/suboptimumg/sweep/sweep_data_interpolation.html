def check_value_in_axis_range(axis: NumericAxis, value: float | str) -> float:
    """Ensure a value is a number within an axis's range, allowing for rounding."""
    if isinstance(value, (str, bool)):
        raise ValueError(
            f"`{axis.name}` is numeric, so it must be fixed at a number, got {value!r}"
        )
    low, high = axis.values[0], axis.values[-1]
    if not (low <= value <= high or np.isclose(value, low) or np.isclose(value, high)):
        raise ValueError(
            f"Can't fix `{axis.name}` at {value}: it is outside the swept range [{low}, {high}]"
        )
    return float(value)

def interpolate_with_axis_fixed(
    axes: list[NumericAxis | CategoricalAxis],
    values: NDArray[np.float64],
    dimension: int,
    value: float,
) -> NDArray[np.float64]:
    """
    Interpolate gridded values with one numeric axis fixed, over the grid of the remaining axes.

    Parameters
    ----------
    axes : list[NumericAxis | CategoricalAxis]
        Axes of the grid, in order of the array dimensions
    values : NDArray[np.float64]
        Values shaped like the grid
    dimension : int
        Dimension of the fixed axis, which must be numeric
    value : float
        Value of the fixed axis

    Returns
    -------
    NDArray[np.float64]
        Values shaped like the grid of the remaining axes. Points that depend on a failed (`NaN`) point are `NaN`.

    Notes
    -----
    Values are never interpolated across a categorical axis. For a grid with C categorical and N numeric axes:

    1. The categorical dimensions are moved to the front, so the array is shaped
       (categorical sizes..., numeric sizes...).
    2. Each combination of category indices selects one N-D numeric sub-grid. That sub-grid is interpolated on its
       own with `RegularGridInterpolator`, with the fixed axis set to `value`, giving an (N-1)-D result.
    3. The results are stacked into a (C + N - 1)-D array, shaped (categorical sizes..., free numeric sizes...).
    4. The categorical dimensions are moved back to their original order relative to the free numeric axes.

    For example, axes (mass, motor, ratio) shaped (5, 3, 31) with mass fixed: the array is reordered to (3, 5, 31),
    each motor's (5, 31) grid is interpolated at the fixed mass to give 31 values, the stack is (3, 31), and the
    result keeps the order (motor, ratio).
    """
    categorical_dimensions = [
        other for other, axis in enumerate(axes) if isinstance(axis, CategoricalAxis)
    ]
    numeric_dimensions = [other for other, axis in enumerate(axes) if isinstance(axis, NumericAxis)]
    numeric_axes = [axis for axis in axes if isinstance(axis, NumericAxis)]
    fixed_dimension_among_numeric_axes = numeric_dimensions.index(dimension)

    category_count = len(categorical_dimensions)
    categories_first = np.moveaxis(values, categorical_dimensions, range(category_count))
    category_shape = categories_first.shape[:category_count]
    free_numeric_shape = tuple(
        len(axis.values)
        for position, axis in enumerate(numeric_axes)
        if position != fixed_dimension_among_numeric_axes
    )

    reduced = np.empty(category_shape + free_numeric_shape, dtype=np.float64)
    for category_index in np.ndindex(category_shape):
        reduced[category_index] = _interpolate_numeric_grid_with_axis_fixed(
            numeric_axes,
            categories_first[category_index],
            fixed_dimension_among_numeric_axes,
            value,
        )

    remaining_dimensions = [other for other in range(len(axes)) if other != dimension]
    destinations = [remaining_dimensions.index(other) for other in categorical_dimensions]
    return np.moveaxis(reduced, range(category_count), destinations)




def _interpolate_numeric_grid_with_axis_fixed(
    axes: list[NumericAxis], values: NDArray[np.float64], dimension: int, value: float
) -> NDArray[np.float64]:
    free_dimensions = [other for other in range(len(axes)) if other != dimension]
    free_shape = tuple(len(axes[other].values) for other in free_dimensions)

    # One grid per free axis, holding that axis's coordinate at every grid cell.
    # e.g. mass = [200, 250, 300], cg_h = [0.25, 0.30], np.meshgrid(mass, cg_h, indexing="ij")
    #  mesh[0] = mass at every cell [[200, 200], [250, 250], [300, 300]]
    #  mesh[1] = cg_h at every cell [[0.25, 0.30], [0.25, 0.30], [0.25, 0.30]]
    mesh = np.meshgrid(*(axes[other].values for other in free_dimensions), indexing="ij")

    points = np.empty((math.prod(free_shape), len(axes)), dtype=np.float64)
    for position, other in enumerate(free_dimensions):
        points[:, other] = mesh[position].ravel()
    points[:, dimension] = value

    interpolator = RegularGridInterpolator(
        tuple(axis.values for axis in axes),
        values,
        method="linear",
        bounds_error=True,
        fill_value=np.nan,
    )
    return interpolator(points).reshape(free_shape)

def estimate_partial_derivatives(
    axes: list[NumericAxis], values: NDArray[np.float64], point: NDArray[np.float64]
) -> dict[str, float]:
    """
    Estimate the partial derivatives of gridded values at one point.

    Parameters
    ----------
    axes : list[NumericAxis]
        Axes of the grid, in order of the array dimensions
    values : NDArray[np.float64]
        Values shaped like the grid
    point : NDArray[np.float64]
        Coordinates of the point, one per axis

    Returns
    -------
    dict[str, float]
        Partial derivative along each axis, keyed by parameter name

    Notes
    -----
    Each partial is a numerical derivative of the interpolated values along one axis, with the other axes held at `point`.
    The initial step is ``DERIVATIVE_INITIAL_STEP_SIZE`` of the axis's span.
    """
    interpolator = RegularGridInterpolator(
        tuple(axis.values for axis in axes),
        values,
        method="linear",
        bounds_error=False,
        fill_value=None,
    )
    partial_derivatives: dict[str, float] = {}
    for dimension, axis in enumerate(axes):

        def evaluate_along_dimension(coordinates: NDArray[np.float64], /) -> NDArray[np.float64]:
            # `derivative` passes an array of coordinates along this axis, in any shape
            # One copy of `point` per coordinate, as rows of an (n, number of axes) array
            points = np.tile(point, (np.size(coordinates), 1))
            # Move each copy along this axis to one coordinate, keeping the other axes fixed
            points[:, dimension] = np.ravel(coordinates)
            # Interpolate every row, then return the results in the shape `derivative` passed
            return np.reshape(interpolator(points), np.shape(coordinates))

        result = derivative(
            evaluate_along_dimension,
            point[dimension],
            initial_step=float(axis.values[-1] - axis.values[0]) * DERIVATIVE_INITIAL_STEP_SIZE,
            preserve_shape=True,
        )
        partial_derivatives[axis.name] = float(result.df)
    return partial_derivatives
