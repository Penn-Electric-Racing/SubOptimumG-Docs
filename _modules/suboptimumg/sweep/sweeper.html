class Sweeper:
    """
    N-dimensional parameter sweeper that owns its configuration and results.

    Parameters
    ----------
    comp_data : CompetitionData
        Base competition. It is copied, so later edits to the original have no effect.
    inputs : Sequence[SweepInput]
        Swept parameters (`SweepParamConfig`) and labeled replacements
        (`CategoricalSweepInput`, e.g. motors), one axis each, in this order
    dependencies : Sequence[VehicleDependency], optional
        Functions that take the vehicle model, with a point's swept values
        applied, and return other parameters to change at that point
    executor : SweepExecutor, optional
        How simulations are run (default runs them in a process pool)

    Notes
    -----
    `comp_data` and `config` can be edited after construction. `sweep()` only
    reruns the simulations when the generated configurations have changed.
    """

    def __init__(
        self,
        comp_data: CompetitionData,
        inputs: Sequence[SweepInput],
        dependencies: Sequence[VehicleDependency] = (),
        executor: SweepExecutor = MultiprocessSweepExecutor(),
    ) -> None:
        self.comp_data = comp_data.model_copy(deep=True)
        self.config = SweepConfigGenerator(inputs=list(inputs), dependencies=list(dependencies))
        self.executor = executor
        self._data: SweepData | None = None
        self._fingerprint: str | None = None

    def sweep(self, force: bool = False) -> None:
        """
        Run every simulation of the sweep, unless the results are already up to date.

        Parameters
        ----------
        force : bool, optional
            Rerun even when the results are up to date (default is False)
        """
        plan = self._generate_plan()
        fingerprint = self._compute_plan_fingerprint(plan)
        if not force and fingerprint == self._fingerprint:
            print("Sweep results are up to date; skipping the simulations.")
            return

        results = self.executor.simulate_configurations(self.comp_data, plan.configurations)
        self._data = SweepData.from_results(plan, results)
        self._fingerprint = fingerprint



    @property
    def is_stale(self) -> bool:
        """Whether the configuration has changed since the last sweep, or none has run."""
        return self._fingerprint != self._compute_plan_fingerprint(self._generate_plan())

    @property
    def data(self) -> SweepData:
        """Results of the last sweep."""
        if self._data is None:
            raise RuntimeError("No results yet. Call sweep() first.")
        if self.is_stale:
            print(
                "The sweep configuration has changed since the last sweep(). "
                "These results are out of date; call sweep() to refresh them."
            )
        return self._data

    def plot(
        self,
        metric: SweepDatatype = SweepDatatype.TOTAL_PTS,
        plot_type: PlotType = PlotType.CONTOUR,
        *,
        fixed_params: Mapping[str, float | str] = {},
        title: str | None = None,
        subtitle: str | None = None,
        x_label: str | None = None,
        y_label: str | None = None,
        z_label: str | None = None,
        theme: str = DEFAULT_THEME,
        **plot_options: object,
    ) -> go.Figure:
        """
        Plot one metric against the swept parameters that are left free.

        Parameters
        ----------
        metric : SweepDatatype, optional
            Metric to plot (default is SweepDatatype.TOTAL_PTS)
        plot_type : PlotType, optional
            Contour or surface, for 2 free parameters (default is PlotType.CONTOUR)
        fixed_params : Mapping[str, float | str], optional
            Swept parameters to fix, e.g. `{"mass": 250, "pwrtn.motor": "emrax208"}`.
            Numeric values between grid points are interpolated. Categorical axes
            take one of their labels.
        title, subtitle, x_label, y_label, z_label : str, optional
            Text for the plot. Defaults are generated from the swept parameters.
        theme : str, optional
            Color theme name
        **plot_options : object
            Passed to the plotting function, e.g. `fit_curve=True` or `scene_config=...`

        Returns
        -------
        go.Figure
            A line plot for 1 free numeric parameter, one line per label when a
            categorical axis is also free, or a contour or surface for 2 numeric
        """
        return plot_sweep(
            self.data.fix_params(fixed_params),
            metric=metric,
            plot_type=plot_type,
            title=title,
            subtitle=subtitle,
            x_label=x_label,
            y_label=y_label,
            z_label=z_label,
            theme=theme,
            **plot_options,
        )

    def grid_plot(
        self,
        plot_type: PlotType = PlotType.CONTOUR,
        show_event_times: bool = False,
        *,
        fixed_params: Mapping[str, float | str] = {},
        title: str | None = None,
        x_label: str | None = None,
        y_label: str | None = None,
        rows: int = 2,
        cols: int = 3,
        theme: str = DEFAULT_THEME,
        **plot_options: object,
    ) -> go.Figure:
        """
        Plot every event's metric against the swept parameters that are left free, in a grid.

        Parameters
        ----------
        plot_type : PlotType, optional
            Contour or surface, for 2 free parameters (default is PlotType.CONTOUR)
        show_event_times : bool, optional
            Show event times instead of event points (default is False)
        fixed_params : Mapping[str, float | str], optional
            Swept parameters to fix, e.g. `{"mass": 250, "pwrtn.motor": "emrax208"}`.
            Categorical axes take one of their labels.
        title, x_label, y_label : str, optional
            Text for the plot. Defaults are generated from the swept parameters.
        rows, cols : int, optional
            Shape of the grid (default is 2 by 3)
        theme : str, optional
            Color theme name
        **plot_options : object
            Passed to the plotting function, e.g. `fit_curve=True` or `show_legend=True`

        Returns
        -------
        go.Figure
            A grid with one subplot per metric
        """
        return grid_plot_sweep(
            self.data.fix_params(fixed_params),
            plot_type=plot_type,
            show_event_times=show_event_times,
            title=title,
            x_label=x_label,
            y_label=y_label,
            rows=rows,
            cols=cols,
            theme=theme,
            **plot_options,
        )

    def evaluate_sweep_metric(
        self,
        fixed_params: Mapping[str, float | str],
        metric: SweepDatatype = SweepDatatype.TOTAL_PTS,
    ) -> float:
        """
        Read a metric at one point, interpolating between grid points.

        Parameters
        ----------
        fixed_params : Mapping[str, float | str]
            Value of every swept parameter, e.g. `{"mass": 250, "pwrtn.motor": "emrax208"}`
        metric : SweepDatatype, optional
            Metric to read (default is SweepDatatype.TOTAL_PTS)

        Returns
        -------
        float
            The metric's value, or `NaN` if it depends on a failed point
        """
        return self.data.evaluate_metric_at_point(fixed_params, metric)

    def evaluate_sweep_metric_gradient(
        self,
        fixed_params: Mapping[str, float | str],
        metric: SweepDatatype = SweepDatatype.TOTAL_PTS,
    ) -> SweepGradient:
        """
        Estimate the partial derivatives of a metric at one point.

        Parameters
        ----------
        fixed_params : Mapping[str, float | str]
            Value of every swept parameter, e.g. `{"mass": 250, "pwrtn.motor": "emrax208"}`
        metric : SweepDatatype, optional
            Metric to differentiate (default is SweepDatatype.TOTAL_PTS)

        Returns
        -------
        SweepGradient
            One partial derivative per numeric swept parameter
        """
        return self.data.estimate_metric_gradient(fixed_params, metric)



    def _generate_plan(self) -> SweepPlan:
        return self.config.generate_sweep_plan(self.comp_data.vehicle_model)

    def _compute_plan_fingerprint(self, plan: SweepPlan) -> str:
        return hashlib.sha256(pickle.dumps((self.comp_data, plan.configurations))).hexdigest()
