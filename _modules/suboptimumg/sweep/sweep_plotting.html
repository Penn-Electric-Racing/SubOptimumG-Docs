def plot_sweep(
    data: SweepData,
    metric: SweepMetric = SweepMetric.TOTAL_PTS,
    plot_type: PlotType = PlotType.CONTOUR,
    title: str | None = None,
    subtitle: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    z_label: str | None = None,
    theme: str = DEFAULT_THEME,
    **plot_options: object,
) -> go.Figure:
    """
    Plot one metric of sweep data against the swept axes left free.

    Parameters
    ----------
    data : SweepData
        Sweep data, with 1 or 2 numeric axes free, or 1 categorical and 1 numeric axis free
    metric : SweepMetric, optional
        Metric to plot (default is SweepMetric.TOTAL_PTS)
    plot_type : PlotType, optional
        Contour or surface, used only with 2 free numeric axes (default is PlotType.CONTOUR)
    title, subtitle, x_label, y_label, z_label : str or None, optional
        Text for the plot. None (the default) generates text from the swept parameters.
    theme : str, optional
        Color theme name (default is DEFAULT_THEME)
    **plot_options : object
        Passed to the plotting function, e.g. `fit_curve=True`

    Returns
    -------
    go.Figure
        A line plot for 1 free numeric axis, one line per label when a categorical axis is also free,
        or a contour or surface for 2 free numeric axes
    """
    numeric_axes, categorical_axes = _check_free_axes_are_plottable(data)
    final_title = title or _build_default_title(data)
    final_subtitle = subtitle or _build_default_subtitle(data)
    values = data.metric_values(metric)

    if categorical_axes:
        x_axis = numeric_axes[0]
        return overlay_plot2D(
            x_data_dict={label: x_axis.values for label in categorical_axes[0].labels},
            y_data_dict=_split_values_by_label(data, categorical_axes[0], values),
            title=final_title,
            x_axis=x_label or x_axis.name,
            y_axis=y_label or TITLE_MAPPING[metric],
            subtitle=final_subtitle,
            theme=theme,
            **plot_options,  # type: ignore[arg-type]
        )

    if len(numeric_axes) == 1:
        return plot2D(
            x_list=numeric_axes[0].values,
            y_list=values,
            title=final_title,
            x_axis=x_label or numeric_axes[0].name,
            y_axis=y_label or TITLE_MAPPING[metric],
            subtitle=final_subtitle,
            theme=theme,
            **plot_options,  # type: ignore[arg-type]
        )

    x_axis, y_axis = numeric_axes
    plotter = plot3D_surface if plot_type == PlotType.SURFACE else plot3D_contour
    return plotter(
        x_list=x_axis.values,
        y_list=y_axis.values,
        z_list=values,
        title=final_title,
        x_axis=x_label or x_axis.name,
        y_axis=y_label or y_axis.name,
        z_axis=z_label or TITLE_MAPPING[metric],
        subtitle=final_subtitle,
        theme=theme,
        **plot_options,  # type: ignore[arg-type]
    )

def grid_plot_sweep(
    data: SweepData,
    plot_type: PlotType = PlotType.CONTOUR,
    show_event_times: bool = False,
    title: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    rows: int = 2,
    cols: int = 3,
    theme: str = DEFAULT_THEME,
    **plot_options: object,
) -> go.Figure:
    """
    Plot every event's metric of sweep data against the free swept axes, in a grid.

    Parameters
    ----------
    data : SweepData
        Sweep data, with 1 or 2 numeric axes free, or 1 categorical and 1 numeric axis free
    plot_type : PlotType, optional
        Contour or surface, used only with 2 free numeric axes (default is PlotType.CONTOUR)
    show_event_times : bool, optional
        Show event times instead of event points (default is False)
    title, x_label, y_label : str or None, optional
        Text for the plot. None (the default) generates text from the swept parameters.
    rows, cols : int, optional
        Shape of the grid (default is 2 by 3)
    theme : str, optional
        Color theme name (default is DEFAULT_THEME)
    **plot_options : object
        Passed to the plotting function, e.g. `fit_curve=True` or `show_legend=True`

    Returns
    -------
    go.Figure
        A grid of line plots for 1 free numeric axis, with one line per label when a categorical axis is
        also free, or of contours or surfaces for 2 free numeric axes
    """
    numeric_axes, categorical_axes = _check_free_axes_are_plottable(data)
    metrics = TIME_GRID_PLOT if show_event_times else STANDARD_GRID_PLOT
    subplot_titles = [TITLE_MAPPING[metric] for metric in metrics]
    values_by_metric = {metric.value: data.metric_values(metric) for metric in metrics}

    final_title = title or f"{_build_default_title(data)} Grid Plot"
    subtitle = _build_default_subtitle(data)
    if subtitle:
        final_title += f"<br><span style='{"font-size: 18px; color: #606060;"}'>{subtitle}</span>"

    if categorical_axes:
        x_axis = numeric_axes[0]
        labels = categorical_axes[0].labels
        return overlay_grid_plot_2D(
            x_data_dict={
                metric_name: {label: x_axis.values for label in labels}
                for metric_name in values_by_metric
            },
            y_data_dict={
                metric_name: _split_values_by_label(data, categorical_axes[0], values)
                for metric_name, values in values_by_metric.items()
            },
            subplot_titles=subplot_titles,
            title=final_title,
            x_label=x_label or x_axis.name,
            y_label=y_label or "",
            rows=rows,
            cols=cols,
            theme=theme,
            **plot_options,  # type: ignore[arg-type]
        )

    if len(numeric_axes) == 1:
        return plot_grid_2D(
            x_list=numeric_axes[0].values,
            y_data_dict=values_by_metric,
            subplot_titles=subplot_titles,
            title=final_title,
            x_label=x_label or numeric_axes[0].name,
            y_label=y_label or "",
            rows=rows,
            cols=cols,
            theme=theme,
            **plot_options,  # type: ignore[arg-type]
        )

    x_axis, y_axis = numeric_axes
    return plot_grid_3D(
        x_list=x_axis.values,
        y_list=y_axis.values,
        z_data_dict=values_by_metric,
        subplot_titles=subplot_titles,
        title=final_title,
        x_label=x_label or x_axis.name,
        y_label=y_label or y_axis.name,
        z_label_dict={metric.value: TITLE_MAPPING[metric] for metric in metrics},
        rows=rows,
        cols=cols,
        plot_type=plot_type,
        theme=theme,
        **plot_options,  # type: ignore[arg-type]
    )




def _split_values_by_label(
    data: SweepData, axis: CategoricalAxis, values: NDArray[np.float64]
) -> dict[str, NDArray[np.number]]:
    dimension = next(position for position, other in enumerate(data.axes) if other is axis)
    return {
        label: np.take(values, index, axis=dimension) for index, label in enumerate(axis.labels)
    }


def _check_free_axes_are_plottable(
    data: SweepData,
) -> tuple[list[NumericAxis], list[CategoricalAxis]]:
    """
    Ensure the free axes fit one of the supported plots (1 numeric axis,
    2 numeric axes, or 1 categorical and 1 numeric axis.)

    Parameters
    ----------
    data : SweepData
        Sweep data, after fixing parameters

    Returns
    -------
    tuple[list[NumericAxis], list[CategoricalAxis]]
        The free numeric axes and the free categorical axes, each in array order

    Raises
    ------
    ValueError
        Unplottable free axes configurations
    """
    numeric_axes = [axis for axis in data.axes if isinstance(axis, NumericAxis)]
    categorical_axes = [axis for axis in data.axes if isinstance(axis, CategoricalAxis)]
    free_names = ", ".join(f"`{axis.name}`" for axis in data.axes)

    if not data.axes:
        raise ValueError(
            "Every swept parameter has been fixed. Remove one or two parameters from fixed_params"
        )
    if not numeric_axes:
        raise ValueError(
            f"Only categorical axes ({free_names}) are free, but plots need a numeric axis for x. "
            "Leave one numeric parameter out of fixed_params"
        )
    if len(categorical_axes) > 1 or len(data.axes) > 2:
        raise ValueError(
            "Plots can show at most 2 dimensions, with at most 1 categorical axis, "
            f"but {free_names} are free. Use fixed_params to reduce dimensionality"
        )
    return numeric_axes, categorical_axes


def _build_default_title(data: SweepData) -> str:
    return " and ".join(axis.name for axis in data.axes) + " Sweep"


def _build_default_subtitle(data: SweepData) -> str | None:
    parts: list[str] = []
    if data.dependency_names:
        parts.append(f"Dependent Params: {', '.join(data.dependency_names)}")
    if data.fixed_axes:
        fixed = ", ".join(f"{axis.name} = {axis.get_formatted_value()}" for axis in data.fixed_axes)
        parts.append(f"Fixed Params: {fixed}")
    return " | ".join(parts) or None

def plot_array_sweep(
    data: ArraySweepData,
    metric: SweepMetric = SweepMetric.TOTAL_PTS,
    title: str | None = None,
    y_label: str | None = None,
    **plot_options: object,
) -> go.Figure:
    """
    Plot one metric of every parameter of an array sweep, colored by percent change.
    Failed points are `NaN` and are not drawn; see each parameter's `SweepData.errors`.

    Parameters
    ----------
    data : ArraySweepData
        Results of an array sweep
    metric : SweepMetric, optional
        Metric to plot (default is SweepMetric.TOTAL_PTS)
    title, y_label : str or None, optional
        Text for the plot. None (the default) generates text from the metric.
    **plot_options : object
        Passed to `plot_bar_chart`, e.g. `theme="powertrain"`

    Returns
    -------
    go.Figure
        One column of points per parameter, with the unmodified car's value as a baseline
    """
    return plot_bar_chart(
        data_by_variable=[
            BarChartSeries(var_name=name, points=parameter_data.metric_values(metric))
            for name, parameter_data in data.parameter_data.items()
        ],
        percent_steps=np.array(data.percent_changes),
        baseline_value=float(data.baseline.metric_values(metric)),
        title=title or "Independent Variable Sweep Results",
        y_label=y_label or TITLE_MAPPING[metric],
        **plot_options,  # type: ignore[arg-type]
    )
