def build_competition(base: CompetitionData, configuration: CarConfiguration) -> Competition:
    """
    Build the competition for one configuration.

    Parameters
    ----------
    base : CompetitionData
        The competition the configuration starts from. It is not modified.
    configuration : CarConfiguration
        Changes to apply to the base vehicle, in order

    Returns
    -------
    Competition
        A competition whose car is built from the changed vehicle
    """
    vehicle = base.vehicle_model.model_copy(deep=True)
    for modification in configuration.modifications:
        modification.apply_to_vehicle(vehicle)
    return from_data(base.model_copy(update={"vehicle_model": vehicle}))

def simulate_safely(
    simulator: ConfigurationSimulator, base: CompetitionData, configuration: CarConfiguration
) -> ConfigurationResult:
    """
    Simulate one configuration, capturing what it prints and any error it raises.

    Parameters
    ----------
    simulator : ConfigurationSimulator
        What simulating the configuration means
    base : CompetitionData
        The competition the configuration starts from
    configuration : CarConfiguration
        Changes to apply to the base car

    Returns
    -------
    ConfigurationResult
        The simulator's success with anything printed as warnings, or a `ConfigurationFailure` if it raised
    """
    string_buffer = StringIO()
    old_stdout = sys.stdout
    sys.stdout = string_buffer
    try:
        success = simulator.simulate(base, configuration)
        result: ConfigurationResult = success.model_copy(
            update={"warnings": string_buffer.getvalue()}
        )
    except Exception as error:
        result = ConfigurationFailure(
            error=f"{error}\n{traceback.format_exc()}", warnings=string_buffer.getvalue()
        )
    finally:
        sys.stdout = old_stdout
    return result
