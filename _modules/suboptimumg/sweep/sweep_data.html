class SweepGradient(BaseModel):
    """Partial derivatives of one metric at one point of a sweep."""

    model_config = ConfigDict(frozen=True)

    metric: SweepMetric = Field(description="Metric that was differentiated")
    fixed_params: dict[str, float | str] = Field(
        description="Point of evaluation, keyed by parameter name. Categorical axes hold a label."
    )
    partial_derivatives: dict[str, float] = Field(
        description="Partial derivative along each numeric swept parameter, keyed by parameter name"
    )

    @computed_field  # type: ignore[prop-decorator]
    @property
    def magnitude(self) -> float:
        return math.sqrt(sum(value**2 for value in self.partial_derivatives.values()))

class SweepData(BaseModel):
    """Results of an N-dimensional sweep, one array entry per grid point."""

    model_config = ConfigDict(arbitrary_types_allowed=True, frozen=True)

    axes: list[SweepAxis] = Field(description="Free axes, in order of the array dimensions")
    fixed_axes: list[FixedAxis] = Field(
        default_factory=list, description="Axes removed by fixing them at one value"
    )
    dependency_names: list[str] = Field(description="Every parameter set by a dependency")

    accel_pts: NDArray[np.float64] = Field(description="Acceleration event points")
    skidpad_pts: NDArray[np.float64] = Field(description="Skidpad event points")
    autoX_pts: NDArray[np.float64] = Field(description="Autocross event points")
    endurance_pts: NDArray[np.float64] = Field(description="Endurance event points")
    efficiency_pts: NDArray[np.float64] = Field(description="Efficiency points")
    accel_t: NDArray[np.float64] = Field(description="Acceleration event time")
    skidpad_t: NDArray[np.float64] = Field(description="Skidpad event time")
    autoX_t: NDArray[np.float64] = Field(description="Autocross event time")
    endurance_t: NDArray[np.float64] = Field(description="Endurance event time")
    errors: list[str] = Field(description="One message per failed configuration")

    @computed_field  # type: ignore[prop-decorator]
    @property
    def total_pts(self) -> NDArray[np.float64]:
        """Compute total points as sum of all point categories."""
        return (
            self.accel_pts
            + self.skidpad_pts
            + self.autoX_pts
            + self.endurance_pts
            + self.efficiency_pts
        )

    @classmethod
    def from_results(cls, plan: SweepPlan, results: list[ConfigurationResult]) -> SweepData:
        """
        Arrange executor results onto the sweep grid.

        Parameters
        ----------
        plan : SweepPlan
            The plan whose configurations were executed
        results : list[ConfigurationResult]
            One result per configuration, in the plan's order

        Returns
        -------
        SweepData
            Metrics shaped like the grid. Failed points are `NaN`.
        """
        if len(results) != len(plan.configurations):
            raise ValueError(f"Expected {len(plan.configurations)} results, got {len(results)}")

        shape = tuple(axis.size for axis in plan.axes)
        competition_results = [
            result.results if isinstance(result, ConfigurationSuccess) else None
            for result in results
        ]

        def metric_grid_for_attribute(attribute_path: str) -> NDArray[np.float64]:
            read = attrgetter(attribute_path)
            flat = [np.nan if result is None else read(result) for result in competition_results]
            return np.array(flat, dtype=np.float64).reshape(shape)

        return cls(
            axes=plan.axes,
            dependency_names=plan.dependency_names,
            accel_pts=metric_grid_for_attribute("accel.points"),
            skidpad_pts=metric_grid_for_attribute("skidpad.points"),
            autoX_pts=metric_grid_for_attribute("autoX.points"),
            endurance_pts=metric_grid_for_attribute("endurance.points"),
            efficiency_pts=metric_grid_for_attribute("efficiency_points"),
            accel_t=metric_grid_for_attribute("accel.tyour"),
            skidpad_t=metric_grid_for_attribute("skidpad.tyour"),
            autoX_t=metric_grid_for_attribute("autoX.tyour"),
            endurance_t=metric_grid_for_attribute("endurance.tyour"),
            errors=[
                f"{configuration}: {result.error}"
                for configuration, result in zip(plan.configurations, results)
                if isinstance(result, ConfigurationFailure)
            ],
        )

    def fix_params(self, fixed_params: Mapping[str, float | str]) -> SweepData:
        """
        Remove swept axes by fixing each at one value.

        Parameters
        ----------
        fixed_params : Mapping[str, float | str]
            Value for each axis to fix, keyed by name, e.g. `{"aero.cl": 3.4, "pwrtn.motor": "emrax208"}`.
            A numeric axis takes any number within its swept range. A categorical axis takes one of its labels.

        Returns
        -------
        SweepData
            Data over the axes left free

        Notes
        -----
        Fixing a categorical axis selects its label's slice exactly.
        Numeric values are linearly interpolated within each category. NaN's will affect adjacent cells
        """
        data = self
        for name, value in fixed_params.items():
            data = data._fix_one_param(name, value)
        return data

    def evaluate_metric_at_point(
        self,
        fixed_params: Mapping[str, float | str],
        metric: SweepMetric = SweepMetric.TOTAL_PTS,
    ) -> float:
        """
        Read a metric at one point, interpolating between grid points.

        Parameters
        ----------
        fixed_params : Mapping[str, float | str]
            Value of every swept axis, keyed by name. Categorical axes take a label.
        metric : SweepMetric, optional
            Metric to read (default is SweepMetric.TOTAL_PTS)

        Returns
        -------
        float
            The metric's value, or `NaN` if it depends on a failed point
        """
        self._check_every_param_fixed(fixed_params)
        return float(self.fix_params(fixed_params).metric_values(metric))

    def estimate_metric_gradient(
        self,
        fixed_params: Mapping[str, float | str],
        metric: SweepMetric = SweepMetric.TOTAL_PTS,
    ) -> SweepGradient:
        """
        Estimate the partial derivatives of a metric at one point.

        Parameters
        ----------
        fixed_params : Mapping[str, float | str]
            Value of every swept axis, keyed by name. Categorical axes take a label.
        metric : SweepMetric, optional
            Metric to differentiate (default is SweepMetric.TOTAL_PTS)

        Returns
        -------
        SweepGradient
            One partial derivative per numeric swept parameter

        Notes
        -----
        Categorical axes are fixed first, exactly, and are not differentiated.
        See `estimate_partial_derivatives` for how each numeric partial is computed.
        """
        self._check_every_param_fixed(fixed_params)
        categorical_point = {
            axis.name: fixed_params[axis.name]
            for axis in self.axes
            if isinstance(axis, CategoricalAxis)
        }
        numeric_data = self.fix_params(categorical_point)
        numeric_axes = [axis for axis in numeric_data.axes if isinstance(axis, NumericAxis)]
        numeric_point = {
            axis.name: check_value_in_axis_range(axis, fixed_params[axis.name])
            for axis in numeric_axes
        }

        partial_derivatives: dict[str, float] = {}
        if numeric_axes:
            partial_derivatives = estimate_partial_derivatives(
                numeric_axes,
                numeric_data.metric_values(metric),
                np.array(list(numeric_point.values()), dtype=np.float64),
            )
        evaluated_at: dict[str, float | str] = {**categorical_point, **numeric_point}
        return SweepGradient(
            metric=metric,
            fixed_params={axis.name: evaluated_at[axis.name] for axis in self.axes},
            partial_derivatives=partial_derivatives,
        )



    def _try_get_dimension_of_param(self, name: str) -> int:
        for dimension, axis in enumerate(self.axes):
            if axis.name == name:
                return dimension
        raise ValueError(
            f"`{name}` is not a free swept parameter. "
            f"Free parameters: {', '.join(f'`{axis.name}`' for axis in self.axes) or '(none)'}"
        )

    def _check_every_param_fixed(self, fixed_params: Mapping[str, float | str]) -> None:
        for name in fixed_params:
            self._try_get_dimension_of_param(name)
        not_fixed = [f"`{axis.name}`" for axis in self.axes if axis.name not in fixed_params]
        if not_fixed:
            raise ValueError(
                "Reading a single point needs every swept parameter in fixed_params. "
                f"Not fixed: {', '.join(not_fixed)}"
            )

    def _fix_one_param(self, name: str, value: float | str) -> SweepData:
        dimension = self._try_get_dimension_of_param(name)
        axis = self.axes[dimension]
        match axis:
            case CategoricalAxis():
                return self._fix_categorical_axis(dimension, axis, value)
            case NumericAxis():
                return self._fix_numeric_axis(dimension, axis, value)

    def _fix_categorical_axis(
        self, dimension: int, axis: CategoricalAxis, value: float | str
    ) -> SweepData:
        index = axis.try_get_index_of_label(value)
        reduced_dimension_metrics = {
            metric.value: np.take(getattr(self, metric.value), index, axis=dimension)
            for metric in SweepMetric
            if metric is not SweepMetric.TOTAL_PTS
        }
        return self._copy_with_axis_fixed_and_reduced_metrics(
            axis,
            FixedAxis(name=axis.name, value=axis.labels[index]),
            reduced_dimension_metrics,
        )

    def _fix_numeric_axis(self, dimension: int, axis: NumericAxis, value: float | str) -> SweepData:
        number = check_value_in_axis_range(axis, value)
        reduced_dimension_metrics = {
            metric.value: interpolate_with_axis_fixed(
                self.axes, getattr(self, metric.value), dimension, number
            )
            for metric in SweepMetric
            if metric is not SweepMetric.TOTAL_PTS
        }
        return self._copy_with_axis_fixed_and_reduced_metrics(
            axis, FixedAxis(name=axis.name, value=number), reduced_dimension_metrics
        )

    def _copy_with_axis_fixed_and_reduced_metrics(
        self,
        axis: NumericAxis | CategoricalAxis,
        fixed_axis: FixedAxis,
        reduced_dimension_metrics: dict[str, NDArray[np.float64]],
    ) -> SweepData:
        return self.model_copy(
            update={
                "axes": [other for other in self.axes if other is not axis],
                "fixed_axes": self.fixed_axes + [fixed_axis],
                **{
                    name: np.asarray(values, dtype=np.float64)
                    for name, values in reduced_dimension_metrics.items()
                },
            }
        )

class ArraySweepData(BaseModel):
    """Results of an array sweep: the unmodified car, and a 1-D sweep of each parameter."""

    model_config = ConfigDict(frozen=True)

    percent_changes: list[float] = Field(
        description="Changes from each parameter's base value, shared by every parameter"
    )
    baseline: SweepData = Field(description="Results of the unmodified car, with no axes")
    parameter_data: dict[str, SweepData] = Field(
        description="1-D results of each swept parameter, in absolute units, keyed by name"
    )

    @classmethod
    def from_results(
        cls, plan: ArraySweepPlan, results: list[ConfigurationResult]
    ) -> ArraySweepData:
        """
        Split executor results back into the baseline and each parameter's sweep.

        Parameters
        ----------
        plan : ArraySweepPlan
            The plan whose configurations were executed
        results : list[ConfigurationResult]
            One result per configuration, in the plan's order

        Returns
        -------
        ArraySweepData
        """
        remaining = iter(results)

        def take_results_for(sweep_plan: SweepPlan) -> SweepData:
            taken = list(itertools.islice(remaining, len(sweep_plan.configurations)))
            return SweepData.from_results(sweep_plan, taken)

        return cls(
            percent_changes=plan.percent_changes,
            baseline=take_results_for(plan.baseline),
            parameter_data={
                name: take_results_for(sweep_plan)
                for name, sweep_plan in plan.parameter_plans.items()
            },
        )
