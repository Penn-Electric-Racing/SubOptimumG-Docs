class WindowMetrics(BaseModel):
    """
    Prediction error for one sub-window. "AC" metrics ignore any constant offset between measured and predicted yaw
    rate and score only how well the prediction tracks its ups and downs.
    """

    subwindow_id: int = Field(description="sw_id of the sub-window")
    duration_s: float = Field(description="Sub-window duration (s)")
    mean_speed_mps: float = Field(description="Mean speed (m/s)")
    dc_offset_rad_s: float = Field(description="Mean residual (rad/s)")
    rmse_raw_rad_s: float = Field(description="RMSE of the raw residual (rad/s)")
    rmse_ac_rad_s: float = Field(description="RMSE of the mean-removed residual (rad/s)")
    yaw_rate_std_meas_rad_s: float = Field(
        description="Standard deviation of the measured yaw rate (rad/s)"
    )
    vaf_ac_pct: float = Field(description="Variance accounted for by the mean-removed fit (%)")

class AggregateMetrics(BaseModel):
    """RMSE and VAF pooled across sub-windows."""

    rmse_raw_rad_s: float = Field(
        description="Root-mean-square of the per-window raw RMSEs (rad/s)"
    )
    rmse_ac_rad_s: float = Field(description="Root-mean-square of the per-window AC RMSEs (rad/s)")
    vaf_ac_pct: float = Field(description="Mean per-window AC VAF (%)")

    @classmethod
    def from_rows(cls, rows: list[WindowMetrics]) -> AggregateMetrics:
        """Pool per-window metrics."""
        return cls(
            rmse_raw_rad_s=float(np.sqrt(np.mean([r.rmse_raw_rad_s**2 for r in rows]))),
            rmse_ac_rad_s=float(np.sqrt(np.mean([r.rmse_ac_rad_s**2 for r in rows]))),
            vaf_ac_pct=float(np.mean([r.vaf_ac_pct for r in rows])),
        )

def vaf_pct(y_meas: NDArray[np.float64], err: NDArray[np.float64]) -> float:
    """Variance of ``y_meas`` accounted for, given residual ``err`` (%)."""
    return float(100.0 * (1.0 - np.var(err) / max(float(np.var(y_meas)), 1e-12)))

def window_metrics(w: WindowArrays, y_hat: NDArray[np.float64]) -> WindowMetrics:
    """Residual diagnostics for one sub-window's prediction."""
    err = w.yaw_rate - y_hat
    dc = float(err.mean())
    return WindowMetrics(
        subwindow_id=w.sw_id,
        duration_s=w.duration_s,
        mean_speed_mps=float(w.speed.mean()),
        dc_offset_rad_s=dc,
        rmse_raw_rad_s=float(np.sqrt(np.mean(err**2))),
        rmse_ac_rad_s=float(np.sqrt(np.mean((err - dc) ** 2))),
        yaw_rate_std_meas_rad_s=float(np.std(w.yaw_rate)),
        vaf_ac_pct=vaf_pct(w.yaw_rate, err - dc),
    )

def band_vaf_pct(
    w: WindowArrays, y_hat: NDArray[np.float64], band_hz: tuple[float, float]
) -> float:
    """
    VAF after zero-phase bandpassing measurement and prediction to ``band_hz``, isolating the yaw-mode frequencies.

    Parameters
    ----------
    w : WindowArrays
        Sub-window with the measured yaw rate.
    y_hat : NDArray[np.float64]
        Predicted yaw rate (rad/s).
    band_hz : tuple[float, float]
        Passband (Hz).

    Returns
    -------
    float
        Band-limited VAF (%).
    """
    fs = 1.0 / float(np.median(np.diff(w.t)))
    sos = butter(4, band_hz, btype="band", fs=fs, output="sos")
    y_meas = sosfiltfilt(sos, w.yaw_rate)
    return vaf_pct(y_meas, y_meas - sosfiltfilt(sos, y_hat))

def information_criteria(rss: float, n_samples: int, n_params: int) -> tuple[float, float]:
    """Gaussian-likelihood ``(AIC, BIC)``, up to a constant shared by every model on the same data."""
    log_mse = np.log(max(rss, 1e-12) / n_samples)
    return float(2.0 * n_params + n_samples * log_mse), float(
        n_params * np.log(n_samples) + n_samples * log_mse
    )

def predict_yaw(
    w: WindowArrays,
    x: NDArray[np.float64],
    model_configuration: ModelConfig,
    layout: ParamLayout,
    irl_car: IrlCar,
) -> NDArray[np.float64]:
    """
    Predict one sub-window's yaw rate at parameter vector ``x``, starting from the measured initial yaw rate.

    The input is offset (``u - u_off``), optionally lagged for front relaxation, then run
    through the LPV bicycle with each axle's stiffness optionally decayed against its measured slip angle.

    Parameters
    ----------
    w : WindowArrays
        Sub-window to predict.
    x : NDArray[np.float64]
        Parameter vector, aligned with ``layout.params``.
    model_configuration : ModelConfig
        Model extensions to apply.
    layout : ParamLayout
        Maps ``x`` onto named parameters.
    irl_car : IrlCar
        Source of mass and geometry.

    Returns
    -------
    NDArray[np.float64]
        Predicted yaw rate (rad/s).
    """
    v = layout.values(x)
    chassis = layout.chassis(x)
    u = w.u_rad - v["u_off"]
    if model_configuration.relaxation == "front":
        u = apply_relaxation_lag(u, w.speed, w.t, v["L_f"])

    Ca_f = Ca_r = None
    if model_configuration.ca_scheduler == "abs_alpha_per_axle":
        slip_f, slip_r = axle_slip_angles(
            w.bicycle_steer_rad, w.body_slip_rad, w.yaw_rate, w.speed, irl_car
        )
        Ca_f = decayed_stiffness(chassis.Ca_f, v["alpha_f"], v["p"], slip_f)
        Ca_r = decayed_stiffness(chassis.Ca_r, v["alpha_r"], v["p"], slip_r)

    return simulate_lpv_yaw(
        w.t, u, w.speed, chassis, irl_car, float(w.yaw_rate[0]), Ca_f=Ca_f, Ca_r=Ca_r
    )

def predict_windows(
    window_arrays: list[WindowArrays],
    x: NDArray[np.float64],
    model_configuration: ModelConfig,
    layout: ParamLayout,
    irl_car: IrlCar,
) -> Prediction:
    """
    Predict every sub-window, unmirrored, at ``x`` and bundle the residual diagnostics.

    Parameters
    ----------
    window_arrays : list[WindowArrays]
        Sub-windows to predict.
    x : NDArray[np.float64]
        Parameter vector.
    model_configuration : ModelConfig
        Model extensions to apply.
    layout : ParamLayout
        Maps ``x`` onto named parameters.
    irl_car : IrlCar
        Source of mass and geometry.

    Returns
    -------
    Prediction
        Predictions, residuals and metrics.
    """
    y_hats = [predict_yaw(w, x, model_configuration, layout, irl_car) for w in window_arrays]
    errs = [w.yaw_rate - y_hat for w, y_hat in zip(window_arrays, y_hats)]
    n_samples = sum(len(e) for e in errs)
    aic, bic = information_criteria(
        sum(float(np.sum(e**2)) for e in errs), n_samples, len(layout.params)
    )
    return Prediction(
        y_hat_per_window=y_hats,
        err_per_window=errs,
        window_metrics=[window_metrics(w, y_hat) for w, y_hat in zip(window_arrays, y_hats)],
        n_samples=n_samples,
        aic=aic,
        bic=bic,
    )

class YawFit(BaseModel):
    """A fitted yaw-response model. Converged parameters and training diagnostics."""

    model_config = ConfigDict(arbitrary_types_allowed=True)

    model_configuration: ModelConfig = Field(description="Model structure that was fit")
    layout: ParamLayout = Field(description="Parameter vector layout x_hat is aligned with")
    irl_car: IrlCar = Field(description="Car whose mass and geometry the fit was run against")
    x_hat: NDArray[np.float64] = Field(description="Fitted parameter vector")
    train: Prediction = Field(description="Predictions and metrics on the training sub-windows")
    final_cost: float = Field(description="least_squares final cost, 0.5 * sum(residual^2)")
    n_eval: int = Field(description="Number of residual evaluations")
    status: int = Field(description="least_squares termination status")
    message: str = Field(description="least_squares termination message")

    @property
    def values(self) -> dict[ParamName, float]:
        """Every fitted parameter, by name."""
        return self.layout.values(self.x_hat)

    def __str__(self) -> str:
        lines = [
            f"[{self.model_configuration}]  final cost = {self.final_cost:.4f}, n_eval = {self.n_eval}, status = {self.status} ({self.message})",
            f"  n_samples = {self.train.n_samples}, n_params = {len(self.layout.params)}, AIC = {self.train.aic:.1f}, BIC = {self.train.bic:.1f}",
            f"  {'param':>8s}  {'init':>16s}  {'fit':>16s}  {'delta':>10s}",
        ]

        def fmt(name: ParamName, value: float) -> str:
            return f"{np.rad2deg(value):.3f} deg" if name == "u_off" else f"{value:.4f}"

        for name, p in self.layout.params.items():
            fitted = self.values[name]
            if abs(p.init) < 1e-9:
                delta = "n/a"
            else:
                delta = f"{100.0 * (fitted - p.init) / abs(p.init):+7.1f}%"
            lines.append(
                f"  {name:>8s}  {fmt(name, p.init):>16s}  {fmt(name, fitted):>16s}  {delta:>10s}"
            )
        return "\n".join(lines)

def fit_yaw(
    window_arrays: list[WindowArrays],
    model_configuration: ModelConfig,
    layout: ParamLayout,
    irl_car: IrlCar,
    verbose: Literal[0, 1, 2] = 0,
    max_nfev: int = 350,
) -> YawFit:
    """
    Fit ``model_configuration`` to the sub-windows by bounded output-error least squares on yaw rate.

    Parameters
    ----------
    window_arrays : list[WindowArrays]
        Training sub-windows.
    model_configuration : ModelConfig
        Variant to fit.
    layout : ParamLayout
        Parameter vector and bounds.
    irl_car : IrlCar
        Source of mass and geometry, recorded on the fit.
    verbose : Literal[0, 1, 2]
        Forwarded to ``scipy.optimize.least_squares``.
    max_nfev : int
        Maximum residual evaluations.

    Returns
    -------
    YawFit
        Converged parameters and training diagnostics.
    """

    def residual(x: NDArray[np.float64]) -> NDArray[np.float64]:
        return np.concatenate(
            [
                w.yaw_rate - predict_yaw(w, x, model_configuration, layout, irl_car)
                for w in window_arrays
            ]
        )

    result = least_squares(
        residual,
        layout.x0,
        bounds=layout.bounds,
        x_scale=layout.x_scale,
        method="trf",
        verbose=verbose,
        max_nfev=max_nfev,
    )
    return YawFit(
        model_configuration=model_configuration,
        layout=layout,
        irl_car=irl_car,
        x_hat=result.x,
        train=predict_windows(window_arrays, result.x, model_configuration, layout, irl_car),
        final_cost=float(result.cost),
        n_eval=int(result.nfev),
        status=int(result.status),
        message=str(result.message),
    )

def validate_fit(val_window_arrays: list[WindowArrays], fit: YawFit) -> Prediction:
    """Evaluate a frozen fit on held-out sub-windows without re-optimizing."""
    return predict_windows(
        val_window_arrays, fit.x_hat, fit.model_configuration, fit.layout, fit.irl_car
    )
