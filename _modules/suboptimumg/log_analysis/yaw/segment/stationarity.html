class StationarityCell(BaseModel):
    """One (window length, tolerance) cell of the stationarity scan."""

    window_s: float = Field(description="Sliding-window length tested (s)")
    tol_pct: float = Field(
        description="Allowed peak-to-peak speed variation, as a percent of the window mean"
    )
    n_windows: int = Field(description="Number of sliding windows evaluated")
    n_passing: int = Field(description="Number of windows whose speed held within tol_pct")
    fraction_passing: float = Field(
        description="n_passing / n_windows; NaN when the window is longer than the log"
    )

class StationarityReport(BaseModel):
    """Grid of ``StationarityCell``, one per (window length, tolerance) pair."""

    speed_col: str = Field(description="Speed channel the scan was run on")
    overlap: float = Field(description="Sliding-window overlap fraction")
    cells: list[StationarityCell] = Field(
        description="One cell per (window length, tolerance) pair"
    )

    def __str__(self) -> str:
        lines = [f"Stationarity scan on {self.speed_col} (overlap={self.overlap:.0%})"]
        lines += [
            f"  len={c.window_s:>5g}s  tol=+/-{c.tol_pct:>4g}%  steady={c.fraction_passing:6.1%}  ({c.n_passing}/{c.n_windows})"
            for c in self.cells
        ]
        return "\n".join(lines)

def scan_speed_stationarity(
    data: SingleRunData,
    win_lengths_s: tuple[float, ...] = (3.0, 5.0, 8.0, 12.0, 20.0),
    tol_pcts: tuple[float, ...] = (2.5, 5.0, 10.0, 15.0),
    speed_col: str = "pcm.vnav.velocityBody.x",
    overlap: float = 0.5,
) -> StationarityReport:
    """
    Measure how often the car holds an approximately constant speed.

    Near-zero steady fractions at long windows show a nonparametric Welch FRF is not viable,
    which is why the yaw response is identified with a parametric LPV fit instead.

    Parameters
    ----------
    data : SingleRunData
        Log to scan.
    win_lengths_s : tuple[float, ...]
        Sliding-window lengths to test (s).
    tol_pcts : tuple[float, ...]
        Allowed peak-to-peak speed variation, as a percent of the window mean.
    speed_col : str
        Speed channel (m/s).
    overlap : float
        Window overlap fraction (0 = none, 0.5 = 50%).

    Returns
    -------
    StationarityReport
        Steady fraction for every (window length, tolerance) pair.
    """
    t_s = timestamps_s(data[speed_col], data.timestamp_unit)
    v = data[speed_col].value_np.astype(np.float64)
    fs = estimate_fs(t_s)
    if fs <= 0:
        raise ValueError("Could not estimate sample rate from the speed timestamps.")

    cells: list[StationarityCell] = []
    for window_s in win_lengths_s:
        n_win = max(2, int(round(window_s * fs)))
        step = max(1, int(round(n_win * (1.0 - overlap))))
        starts = np.arange(0, len(v) - n_win + 1, step)

        if len(starts) == 0:
            cells += [
                StationarityCell(
                    window_s=window_s,
                    tol_pct=tol,
                    n_windows=0,
                    n_passing=0,
                    fraction_passing=float("nan"),
                )
                for tol in tol_pcts
            ]
            continue

        means = np.array([v[s : s + n_win].mean() for s in starts])
        ranges = np.array([np.ptp(v[s : s + n_win]) for s in starts])
        relative_pp = 100.0 * ranges / np.where(np.abs(means) < 1e-3, np.nan, means)

        for tol in tol_pcts:
            n_passing = int(np.sum(relative_pp <= tol))
            cells.append(
                StationarityCell(
                    window_s=window_s,
                    tol_pct=tol,
                    n_windows=len(starts),
                    n_passing=n_passing,
                    fraction_passing=n_passing / len(starts),
                )
            )

    return StationarityReport(speed_col=speed_col, overlap=overlap, cells=cells)
