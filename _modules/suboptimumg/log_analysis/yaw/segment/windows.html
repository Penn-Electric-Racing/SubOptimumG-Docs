class MaskReport(BaseModel):
    """How ``auto_segment`` split one manual window into accepted and too-short sub-windows."""

    manual_window: TimeSpan = Field(
        description="Operator-picked coarse window (t_start, t_end) (s)"
    )
    n_input_pts: int = Field(description="Samples inside the manual window")
    n_kept_pts: int = Field(description="Samples surviving the speed and slip masks")
    n_low_speed_pts: int = Field(description="Samples rejected for being below the speed floor")
    n_high_slip_pts: int = Field(description="Samples rejected for exceeding the slip ceiling")
    sub_windows: list[TimeSpan] = Field(description="Accepted sub-window spans (s)")
    rejected_short: list[TimeSpan] = Field(
        description="Surviving spans dropped for being shorter than min_window_s (s)"
    )

    def __str__(self) -> str:
        w0, w1 = self.manual_window
        return (
            f"window ({w0:.1f}, {w1:.1f})s: {len(self.sub_windows)} kept, {len(self.rejected_short)} too short, "
            f"low-speed={self.n_low_speed_pts}, high-slip={self.n_high_slip_pts}"
        )




def _contiguous_runs(mask: NDArray[np.bool_]) -> list[tuple[int, int]]:
    """``[start, end)`` index pairs for each run of True in ``mask``."""
    edges = np.diff(np.concatenate(([0], mask.astype(np.int8), [0])))
    return list(zip(np.flatnonzero(edges == 1).tolist(), np.flatnonzero(edges == -1).tolist()))


def _mask_window(
    t_s: NDArray[np.float64],
    v: NDArray[np.float64],
    slip: NDArray[np.float64],
    window: TimeSpan,
    min_speed_mps: float,
    max_slip_deg: float,
    min_window_s: float,
) -> MaskReport:
    """Mask one manual window and classify each surviving run as accepted or too short."""
    in_window = (t_s >= window[0]) & (t_s <= window[1])
    t_w, v_w, slip_w = t_s[in_window], v[in_window], slip[in_window]

    low_speed = v_w < min_speed_mps
    high_slip = np.abs(slip_w) > max_slip_deg
    valid = ~(low_speed | high_slip | np.isnan(v_w) | np.isnan(slip_w))

    accepted: list[TimeSpan] = []
    rejected: list[TimeSpan] = []
    for start, end in _contiguous_runs(valid):
        span = (float(t_w[start]), float(t_w[end - 1]))
        (accepted if span[1] - span[0] >= min_window_s else rejected).append(span)

    return MaskReport(
        manual_window=window,
        n_input_pts=int(in_window.sum()),
        n_kept_pts=int(valid.sum()),
        n_low_speed_pts=int(low_speed.sum()),
        n_high_slip_pts=int(high_slip.sum()),
        sub_windows=accepted,
        rejected_short=rejected,
    )

def auto_segment(
    data: SingleRunData,
    manual_windows: list[TimeSpan],
    min_speed_mps: float = 4.0,
    max_slip_deg: float = 7.0,
    min_window_s: float = 8.0,
    speed_col: str = "pcm.vnav.velocityBody.x",
    slip_col: str = BODY_SLIP_ANGLE,
) -> tuple[list[SingleRunData], list[MaskReport]]:
    """
    Split each manual window into clean sub-windows by masking out low-speed and high-slip samples.

    Low speed blows up every 1/V term of the bicycle model, and high body slip leaves the tires' linear region.
    Surviving runs shorter than ``min_window_s`` carry too little low-frequency content to identify anything and are dropped.

    Parameters
    ----------
    data : SingleRunData
        Log with ``speed_col`` and ``slip_col`` channels.
    manual_windows : list[TimeSpan]
        Coarse (t_start, t_end) windows (s), typically read off the GPS trajectory.
    min_speed_mps : float
        Speed floor (m/s).
    max_slip_deg : float
        Body-slip ceiling (deg).
    min_window_s : float
        Shortest sub-window kept (s).
    speed_col : str
        Speed channel (m/s).
    slip_col : str
        Body-slip channel (deg).

    Returns
    -------
    tuple[list[SingleRunData], list[MaskReport]]
        Every accepted sub-window across all manual windows, and one report per manual window.
    """
    speed_di, slip_di = left_join_data_instances(data[speed_col], data[slip_col])
    t_s = timestamps_s(speed_di, data.timestamp_unit)
    v = speed_di.value_np.astype(np.float64)
    slip = slip_di.value_np.astype(np.float64)

    reports = [
        _mask_window(t_s, v, slip, w, min_speed_mps, max_slip_deg, min_window_s)
        for w in manual_windows
    ]
    subwindows = [
        data.trim(
            float(from_seconds(t0, data.timestamp_unit)),
            float(from_seconds(t1, data.timestamp_unit)),
        )
        for report in reports
        for t0, t1 in report.sub_windows
    ]
    return subwindows, reports

class SubwindowSummary(BaseModel):
    """Per-sub-window quality metrics, used to triage windows before fitting."""

    sub_index: int = Field(description="Position of the sub-window in its parent list")
    duration_s: float = Field(description="Sub-window duration (s)")
    fs_hz: float = Field(description="Estimated sample rate (Hz)")
    speed_mean: float = Field(description="Mean speed (m/s)")
    speed_pp_pct: float = Field(description="Peak-to-peak speed variation as a percent of the mean")
    ax_rms: float = Field(description="RMS longitudinal acceleration (m/s^2)")
    input_pp: float = Field(description="Peak-to-peak steering excitation (deg)")
    slip_rms: float = Field(description="RMS body slip angle (deg)")

    def __str__(self) -> str:
        return (
            f"sub {self.sub_index:>2d}: {self.duration_s:5.1f}s @ {self.fs_hz:4.0f}Hz  "
            f"V={self.speed_mean:5.1f} (+/-{self.speed_pp_pct:4.0f}%)  |ax|rms={self.ax_rms:4.2f}  "
            f"steer_pp={self.input_pp:5.1f}  slip_rms={self.slip_rms:4.2f}"
        )




def _rms(x: NDArray[np.float64]) -> float:
    """Root-mean-square, ignoring NaNs."""
    return float(np.sqrt(np.nanmean(x**2)))

def summarize_subwindows(
    subwindows: list[SingleRunData],
    ax_col: str = BODY_LON_ACC,
    input_col: str = "ludwig.steeringWheel.angle",
    speed_col: str = "pcm.vnav.velocityBody.x",
    slip_col: str = BODY_SLIP_ANGLE,
) -> list[SubwindowSummary]:
    """
    Summarize each sub-window's length, speed hold, acceleration, steering excitation and slip.

    Parameters
    ----------
    subwindows : list[SingleRunData]
        Sub-windows to summarize; each must carry all four channels.
    ax_col : str
        Longitudinal acceleration channel (m/s^2).
    input_col : str
        Steering channel (deg).
    speed_col : str
        Speed channel (m/s).
    slip_col : str
        Body slip channel (deg).

    Returns
    -------
    list[SubwindowSummary]
        One summary per sub-window.
    """
    summaries: list[SubwindowSummary] = []
    for i, sub in enumerate(subwindows):
        t_s = timestamps_s(sub[speed_col], sub.timestamp_unit)
        speed = sub[speed_col].value_np.astype(np.float64)
        steer = sub[input_col].value_np.astype(np.float64)
        speed_mean = float(np.nanmean(speed))
        summaries.append(
            SubwindowSummary(
                sub_index=i,
                duration_s=float(t_s[-1] - t_s[0]),
                fs_hz=estimate_fs(t_s),
                speed_mean=speed_mean,
                speed_pp_pct=100.0 * float(np.nanmax(speed) - np.nanmin(speed)) / abs(speed_mean),
                ax_rms=_rms(sub[ax_col].value_np.astype(np.float64)),
                input_pp=float(np.nanmax(steer) - np.nanmin(steer)),
                slip_rms=_rms(sub[slip_col].value_np.astype(np.float64)),
            )
        )
    return summaries
