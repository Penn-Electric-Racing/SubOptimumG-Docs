class ModelConfig(BaseModel):
    """Choice of yaw model to fit: the linear bicycle model, plus either or both of two effects it ignores."""

    model_config = ConfigDict(frozen=True)

    ca_scheduler: CaScheduler = Field(
        default="none",
        description="Tire saturation. 'abs_alpha_per_axle' lowers each axle's cornering stiffness as its slip angle grows, Ca / (1 + alpha |slip|^p), so the model can follow the car near the grip limit; adds alpha_f, alpha_r and p. 'none' keeps stiffness constant.",
    )
    relaxation: Relaxation = Field(
        default="none",
        description="Front tire lag. 'front' passes the steering input through a first-order lag with time constant L_f / V, since a tire must roll some distance before it builds lateral force; adds L_f. 'none' applies steering instantly.",
    )

    def __str__(self) -> str:
        effects = {
            "tire saturation": self.ca_scheduler != "none",
            "front tire lag": self.relaxation != "none",
        }
        return " + ".join(["linear bicycle"] + [effect for effect, on in effects.items() if on])

class Param(BaseModel):
    """Initial guess, bounds and scale of one fitted parameter."""

    model_config = ConfigDict(frozen=True)

    init: float = Field(description="Initial guess")
    lo: float = Field(description="Lower bound")
    hi: float = Field(description="Upper bound")
    scale: float = Field(description="Characteristic magnitude, for the optimizer's x_scale")




DEFAULT_EXTENSION_PARAMS: dict[ParamName, Param] = {
    "alpha_f": Param(init=1.0, lo=0.0, hi=50.0, scale=1.0),
    "alpha_r": Param(init=1.0, lo=0.0, hi=50.0, scale=1.0),
    "p": Param(init=1.0, lo=0.5, hi=4.0, scale=1.0),
    "L_f": Param(init=0.3, lo=0.05, hi=1.5, scale=0.3),
    "u_off": Param(init=0.0, lo=-np.deg2rad(10.0), hi=np.deg2rad(10.0), scale=np.deg2rad(1.0)),

class ParamLayout(BaseModel):
    """The ordered parameter vector of a fit; the optimizer's ``x`` follows ``params`` order."""

    params: dict[ParamName, Param] = Field(description="Every parameter of this fit, in x order")

    @property
    def x0(self) -> NDArray[np.float64]:
        """Initial guess."""
        return np.array([p.init for p in self.params.values()], dtype=np.float64)

    @property
    def bounds(self) -> tuple[NDArray[np.float64], NDArray[np.float64]]:
        """``(lo, hi)`` arrays."""
        return (
            np.array([p.lo for p in self.params.values()], dtype=np.float64),
            np.array([p.hi for p in self.params.values()], dtype=np.float64),
        )

    @property
    def x_scale(self) -> NDArray[np.float64]:
        """Characteristic magnitudes."""
        return np.array([p.scale for p in self.params.values()], dtype=np.float64)

def _param_within_pct_of_prior(prior: float, bound_pct: float) -> Param:
    """A parameter starting at ``prior`` and bounded to +/- ``bound_pct`` percent of it."""
    frac = bound_pct / 100.0
    return Param(init=prior, lo=prior * (1 - frac), hi=prior * (1 + frac), scale=abs(prior))

def build_param_layout(
    model_configuration: ModelConfig, irl_car: IrlCar, overrides: dict[ParamName, Param] = {}
) -> ParamLayout:
    """
    Build the parameter vector for ``model_configuration``, boxed around the car YAML's ``dynamics_setup`` priors.

    ``Ca_f``, ``Ca_r``, ``Izz`` and the steering offset ``u_off`` are always present. The Ca scheduler
    adds per-axle decay rates ``alpha_f``, ``alpha_r`` and a shared exponent ``p``; front relaxation adds ``L_f``.
    Those come from ``DEFAULT_EXTENSION_PARAMS`` unless overridden.

    Parameters
    ----------
    model_configuration : ModelConfig
        Decides which optional parameters are added.
    irl_car : IrlCar
        Source of the chassis priors and their percentage bounds.
    overrides : dict[ParamName, Param]
        Replacement initial guess, bounds and scale for any parameter in ``model_configuration``,
        e.g. ``{"L_f": DEFAULT_EXTENSION_PARAMS["L_f"].model_copy(update={"hi": 2.0})}``.

    Returns
    -------
    ParamLayout
        The parameter vector for ``model_configuration``.
    """
    prior = ChassisParams.from_prior(irl_car)
    pct = irl_car.params.dynamics_setup.prior_bounds
    params: dict[ParamName, Param] = {
        "Ca_f": _param_within_pct_of_prior(prior.Ca_f, pct.Ca_front_pct),
        "Ca_r": _param_within_pct_of_prior(prior.Ca_r, pct.Ca_rear_pct),
        "Izz": _param_within_pct_of_prior(prior.Izz, pct.Izz_pct),
    }
    extensions: list[ParamName] = []
    if model_configuration.ca_scheduler == "abs_alpha_per_axle":
        extensions += ["alpha_f", "alpha_r", "p"]
    if model_configuration.relaxation == "front":
        extensions.append("L_f")
    extensions.append("u_off")
    params |= {name: DEFAULT_EXTENSION_PARAMS[name] for name in extensions}

    unknown = overrides.keys() - params.keys()
    if unknown:
        raise ValueError(
            f"Cannot override {sorted(unknown)}; this model configuration has {list(params)}"
        )
    return ParamLayout(params=params | overrides)
