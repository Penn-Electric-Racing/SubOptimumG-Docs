class ChassisParams(BaseModel):
    """The three chassis scalars that fix the linear bicycle model at every speed."""

    model_config = ConfigDict(frozen=True)

    Ca_f: float = Field(description="Front per-axle cornering stiffness (N/rad)")
    Ca_r: float = Field(description="Rear per-axle cornering stiffness (N/rad)")
    Izz: float = Field(description="Yaw inertia about the CG (kg.m^2)")

class BicycleCoeffs(BaseModel):
    """Coefficients of ``H(s; V) = K wn^2 (1 + T_z s) / (s^2 + 2 zeta wn s + wn^2)``, shaped like the input speeds."""

    model_config = ConfigDict(arbitrary_types_allowed=True)

    K: NDArray[np.float64] = Field(description="DC gain (rad/s per rad steer)")
    T_z: NDArray[np.float64] = Field(description="Numerator (zero) time constant (s)")
    omega_n: NDArray[np.float64] = Field(description="Undamped natural frequency (rad/s)")
    zeta: NDArray[np.float64] = Field(description="Damping ratio")

class BicyclePoint(BaseModel):
    """Linear bicycle coefficients evaluated at a single speed, in display units."""

    V: float = Field(description="Speed (m/s)")
    K: float = Field(description="DC gain (rad/s per rad steer)")
    T_z_ms: float = Field(description="Numerator (zero) time constant (ms)")
    wn_hz: float = Field(description="Undamped natural frequency (Hz)")
    zeta: float = Field(description="Damping ratio")

    def __str__(self) -> str:
        return f"V={self.V:>4.1f}  K={self.K:+.3f}  Tz={self.T_z_ms:+8.1f} ms  wn={self.wn_hz:.2f} Hz  zeta={self.zeta:.3f}"

def cg_split(irl_car: IrlCar) -> tuple[float, float]:
    """CG-to-front and CG-to-rear axle distances (m); ``w_distr_front`` is the front weight fraction."""
    wheelbase = irl_car.params.wb
    lf = wheelbase * (1.0 - irl_car.params.w_distr_front)
    return lf, wheelbase - lf

def bicycle_coeffs(
    V: NDArray[np.float64],
    chassis: ChassisParams,
    irl_car: IrlCar,
    Ca_f: NDArray[np.float64] | None = None,
    Ca_r: NDArray[np.float64] | None = None,
) -> BicycleCoeffs:
    """
    Linear-bicycle (K, T_z, omega_n, zeta) from front tire steer to yaw rate at each speed.

    Every coefficient follows from the three chassis scalars plus known geometry. Free-coefficient LPV fits
    slid along the K/T_z ridge to nonsense on lapping data, so this tie-back is what keeps the fit identifiable.

    Parameters
    ----------
    V : NDArray[np.float64]
        Speeds (m/s), strictly positive.
    chassis : ChassisParams
        Chassis scalars; ``Izz`` is always taken from here.
    irl_car : IrlCar
        Source of mass, wheelbase and CG split.
    Ca_f : NDArray[np.float64] | None
        Per-sample front stiffness overriding ``chassis.Ca_f``, used by the Ca-decay scheduler. None uses ``chassis.Ca_f``.
    Ca_r : NDArray[np.float64] | None
        Per-sample rear stiffness overriding ``chassis.Ca_r``. None uses ``chassis.Ca_r``.

    Returns
    -------
    BicycleCoeffs
        Every field shaped like ``V``.
    """
    V = np.asarray(V, dtype=np.float64)
    if np.any(V <= 0):
        raise ValueError("V must be strictly positive.")

    ca_f = chassis.Ca_f if Ca_f is None else Ca_f
    ca_r = chassis.Ca_r if Ca_r is None else Ca_r
    m, L, Izz = irl_car.params.mass, irl_car.params.wb, chassis.Izz
    lf, lr = cg_split(irl_car)

    omega_n_sq = (ca_f * ca_r * L**2) / (m * Izz * V**2) - (lf * ca_f - lr * ca_r) / Izz
    omega_n_sq = np.maximum(omega_n_sq, 1e-9)
    omega_n = np.sqrt(omega_n_sq)
    two_zeta_wn = (ca_f + ca_r) / (m * V) + (lf**2 * ca_f + lr**2 * ca_r) / (Izz * V)

    return BicycleCoeffs(
        K=(ca_f * ca_r * L) / (m * Izz * V * omega_n_sq),
        T_z=(m * lf * V) / (ca_r * L),
        omega_n=omega_n,
        zeta=two_zeta_wn / (2.0 * omega_n),
    )

def bicycle_points(
    chassis: ChassisParams, irl_car: IrlCar, speeds: tuple[float, ...]
) -> list[BicyclePoint]:
    """Linear bicycle coefficients at each of ``speeds``, in display units."""
    c = bicycle_coeffs(np.asarray(speeds, dtype=np.float64), chassis, irl_car)
    return [
        BicyclePoint(
            V=float(V),
            K=float(c.K[i]),
            T_z_ms=1000.0 * float(c.T_z[i]),
            wn_hz=float(c.omega_n[i]) / (2 * np.pi),
            zeta=float(c.zeta[i]),
        )
        for i, V in enumerate(speeds)
    ]

def bicycle_frequency_response(
    chassis: ChassisParams,
    irl_car: IrlCar,
    V_grid: NDArray[np.float64],
    f_grid: NDArray[np.float64],
) -> NDArray[np.complex128]:
    """
    Complex frequency response ``H(j 2 pi f; V)`` of the linear bicycle over a (speed, frequency) grid.

    Parameters
    ----------
    chassis : ChassisParams
        Chassis scalars.
    irl_car : IrlCar
        Source of mass, wheelbase and CG split.
    V_grid : NDArray[np.float64]
        Speeds (m/s), length nV.
    f_grid : NDArray[np.float64]
        Frequencies (Hz), length nf.

    Returns
    -------
    NDArray[np.complex128]
        Response of shape (nV, nf), in rad/s yaw rate per rad tire steer.
    """
    c = bicycle_coeffs(V_grid, chassis, irl_car)
    K, Tz, wn, zt = (arr[:, None] for arr in (c.K, c.T_z, c.omega_n, c.zeta))
    s = 1j * 2.0 * np.pi * np.asarray(f_grid, dtype=np.float64)[None, :]
    return (K * wn**2) * (1.0 + Tz * s) / (s**2 + 2.0 * zt * wn * s + wn**2)
