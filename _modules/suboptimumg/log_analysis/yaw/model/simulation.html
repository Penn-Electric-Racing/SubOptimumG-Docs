import numpy as np
from numpy.typing import NDArray

from ....vehicle.irl_vehicle import IrlCar
from .bicycle import ChassisParams, bicycle_coeffs, cg_split

V_FLOOR_MPS = 1.0


def _median_time_step(t: NDArray[np.float64]) -> float:
    """Median spacing of ``t`` (s), used as the fixed step of the discrete-time filters below."""
    dt = float(np.median(np.diff(t)))
    if dt <= 0:
        raise ValueError("Time vector must be increasing.")
    return dt

def simulate_lpv_yaw(
    t: NDArray[np.float64],
    u_rad: NDArray[np.float64],
    V: NDArray[np.float64],
    chassis: ChassisParams,
    irl_car: IrlCar,
    y_init: float,
    Ca_f: NDArray[np.float64] | None = None,
    Ca_r: NDArray[np.float64] | None = None,
) -> NDArray[np.float64]:
    """
    Simulate yaw rate with a Tustin biquad whose coefficients are rescheduled at every sample's speed.

    Parameters
    ----------
    t : NDArray[np.float64]
        Uniformly sampled time (s).
    u_rad : NDArray[np.float64]
        Front tire steer input (rad).
    V : NDArray[np.float64]
        Speed at each sample (m/s), floored at ``V_FLOOR_MPS``.
    chassis : ChassisParams
        Chassis scalars.
    irl_car : IrlCar
        Source of mass, wheelbase and CG split.
    y_init : float
        Initial yaw rate (rad/s); the filter state is set so ``y[0]`` equals it exactly.
    Ca_f : NDArray[np.float64] | None
        Per-sample front stiffness (N/rad), or None for the constant ``chassis.Ca_f``.
    Ca_r : NDArray[np.float64] | None
        Per-sample rear stiffness (N/rad), or None for the constant ``chassis.Ca_r``.

    Returns
    -------
    NDArray[np.float64]
        Simulated yaw rate (rad/s), same length as ``t``.
    """
    if not (len(t) == len(u_rad) == len(V)):
        raise ValueError("t, u_rad, V must have the same length.")
    if len(t) < 2:
        return np.full(len(t), y_init, dtype=np.float64)

    c = bicycle_coeffs(np.maximum(V, V_FLOOR_MPS), chassis, irl_car, Ca_f=Ca_f, Ca_r=Ca_r)
    k = 2.0 / _median_time_step(t)
    a0 = c.omega_n**2
    a1 = 2.0 * c.zeta * c.omega_n
    b0 = c.K * a0
    b1 = b0 * c.T_z

    A0 = k * k + a1 * k + a0
    b0_z = (b1 * k + b0) / A0
    b1_z = (2.0 * b0) / A0
    b2_z = (b0 - b1 * k) / A0
    a1_z = (2.0 * a0 - 2.0 * k * k) / A0
    a2_z = (k * k - a1 * k + a0) / A0

    u = np.asarray(u_rad, dtype=np.float64)
    y = np.empty(len(u), dtype=np.float64)
    s1 = y_init - b0_z[0] * u[0]
    s2 = b2_z[0] * u[0] - a2_z[0] * y_init
    for i in range(len(u)):
        y[i] = b0_z[i] * u[i] + s1
        s1 = b1_z[i] * u[i] - a1_z[i] * y[i] + s2
        s2 = b2_z[i] * u[i] - a2_z[i] * y[i]
    return y

def apply_relaxation_lag(
    u_rad: NDArray[np.float64],
    V: NDArray[np.float64],
    t: NDArray[np.float64],
    L_f: float,
) -> NDArray[np.float64]:
    """
    Lag the steering input by a first-order filter with ``tau(V) = L_f / V``, approximating front tire relaxation
    length.

    Parameters
    ----------
    u_rad : NDArray[np.float64]
        Steering input (rad).
    V : NDArray[np.float64]
        Speed at each sample (m/s), floored at ``V_FLOOR_MPS``.
    t : NDArray[np.float64]
        Uniformly sampled time (s).
    L_f : float
        Front relaxation length (m); 0 or less disables the lag.

    Returns
    -------
    NDArray[np.float64]
        Lagged input, same length as ``u_rad``.
    """
    u = np.asarray(u_rad, dtype=np.float64)
    if L_f <= 0 or len(u) < 2:
        return u.copy()

    dt = _median_time_step(t)
    a = dt / (2.0 * L_f / np.maximum(V, V_FLOOR_MPS) + dt)
    y = np.empty_like(u)
    y[0] = u[0]
    for i in range(1, len(u)):
        y[i] = (1.0 - 2.0 * a[i]) * y[i - 1] + a[i] * (u[i] + u[i - 1])
    return y

def axle_slip_angles(
    steer_rad: NDArray[np.float64],
    body_slip_rad: NDArray[np.float64],
    yaw_rate: NDArray[np.float64],
    V: NDArray[np.float64],
    irl_car: IrlCar,
) -> tuple[NDArray[np.float64], NDArray[np.float64]]:
    """
    Front and rear slip-angle magnitudes ``|delta - beta - lf r / V|`` and ``|-beta + lr r / V|`` (rad).

    Parameters
    ----------
    steer_rad : NDArray[np.float64]
        Front bicycle-model tire steer angle (rad).
    body_slip_rad : NDArray[np.float64]
        Body slip angle at the CG (rad).
    yaw_rate : NDArray[np.float64]
        Yaw rate (rad/s).
    V : NDArray[np.float64]
        Speed (m/s), floored at ``V_FLOOR_MPS``.
    irl_car : IrlCar
        Source of the CG split.

    Returns
    -------
    tuple[NDArray[np.float64], NDArray[np.float64]]
        Front and rear slip-angle magnitudes (rad).
    """
    lf, lr = cg_split(irl_car)
    r_over_V = yaw_rate / np.maximum(V, V_FLOOR_MPS)
    return np.abs(steer_rad - body_slip_rad - lf * r_over_V), np.abs(-body_slip_rad + lr * r_over_V)

def decayed_stiffness(
    Ca0: float, decay: float, p: float, slip_rad: NDArray[np.float64]
) -> NDArray[np.float64]:
    """Cornering stiffness ``Ca0 / (1 + decay |alpha|^p)`` after saturating against slip angle ``alpha`` (rad)."""
    return Ca0 / (1.0 + decay * np.power(np.abs(slip_rad), p))
