def bode_speed_grid(
    window_arrays: list[WindowArrays], n: int = 12, floor_mps: float = 4.0
) -> NDArray[np.float64]:
    """Evenly spaced speeds spanning the 5th to 95th percentile of the logged speed, clipped below at ``floor_mps``."""
    speeds = np.concatenate([w.speed for w in window_arrays])
    lo, hi = np.quantile(speeds, [0.05, 0.95])
    return np.linspace(max(float(lo), floor_mps), float(hi), n)

def magnitude_db(H: NDArray[np.complex128]) -> NDArray[np.float64]:
    """Magnitude of ``H`` in dB."""
    return 20.0 * np.log10(np.maximum(np.abs(H), 1e-12))

def phase_deg(H: NDArray[np.complex128]) -> NDArray[np.float64]:
    """Phase of ``H`` in degrees, unwrapped along the frequency (last) axis."""
    return np.rad2deg(np.unwrap(np.angle(H), axis=-1))

def plot_bode_surface(
    fit: YawFit,
    V_grid: NDArray[np.float64],
    f_grid: NDArray[np.float64],
    quantity: BodeQuantity = "magnitude",
) -> go.Figure:
    """
    3D surface of a fit's linear-region frequency response over (speed, log10 frequency).

    Parameters
    ----------
    fit : YawFit
        Supplies the chassis parameters and car.
    V_grid : NDArray[np.float64]
        Speed grid (m/s).
    f_grid : NDArray[np.float64]
        Frequency grid (Hz), positive.
    quantity : BodeQuantity
        Plot magnitude or phase.

    Returns
    -------
    go.Figure
        Rotatable surface.
    """
    H = bicycle_frequency_response(fit.layout.chassis(fit.x_hat), fit.irl_car, V_grid, f_grid)
    if quantity == "phase":
        z, z_label = phase_deg(H), "phase (deg)"
    else:
        z, z_label = magnitude_db(H), "|H| (dB)"

    return plot3D_surface(
        x_list=np.asarray(V_grid, dtype=np.float64),
        y_list=np.log10(f_grid),
        z_list=z,
        title=f"LPV yaw-response {quantity} ({fit.model_configuration})",
        x_axis="speed V (m/s)",
        y_axis="log10 frequency (Hz)",
        z_axis=z_label,
    )

def plot_bode_slices(
    fits: dict[str, YawFit],
    V_picks: list[float],
    f_grid: NDArray[np.float64],
) -> go.Figure:
    """
    Magnitude and phase of one or more fits at fixed speeds; every fit after the first is dashed.

    Parameters
    ----------
    fits : dict[str, YawFit]
        Fits keyed by display name.
    V_picks : list[float]
        Speeds to slice at (m/s).
    f_grid : NDArray[np.float64]
        Frequency grid (Hz).

    Returns
    -------
    go.Figure
        Two-panel Bode plot.
    """
    mag_by: dict[str, NDArray[np.float64]] = {}
    phase_by: dict[str, NDArray[np.float64]] = {}
    dashed: set[str] = set()
    for fit_idx, (name, fit) in enumerate(fits.items()):
        H = bicycle_frequency_response(
            fit.layout.chassis(fit.x_hat), fit.irl_car, np.asarray(V_picks), f_grid
        )
        mag = magnitude_db(H)
        phase = phase_deg(H)
        for v_idx, V in enumerate(V_picks):
            label = f"{name}, V={V:.1f}"
            mag_by[label], phase_by[label] = mag[v_idx], phase[v_idx]
            if fit_idx > 0:
                dashed.add(label)

    return plot_phase_and_magnitude(
        np.asarray(f_grid, dtype=np.float64),
        mag_by,
        phase_by,
        title="LPV yaw-response Bode (linear region, Ca decay = 0)",
        dashed_series=dashed,
    )
